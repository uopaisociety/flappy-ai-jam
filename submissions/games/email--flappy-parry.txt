<script>
// === FLAPPY COMBAT: CONFIG ===
const W = 400, H = 600;
const GRAVITY = 0.45, FLAP_STRENGTH = -7.5;
const PIPE_GAP = 160, PIPE_WIDTH = 64, PIPE_SPEED = 2.6, PIPE_SPACING = 220, GROUND_H = 70;
const COLORS = { navy:"#073B5C", cream:"#F7F5F0", bird:"#FFC93C", beak:"#FF7A00", pipe:"#0E5A8A", ground:"#C9B79C" };
const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");
const statusEl = document.getElementById("status");
let state = "start", bird, pipes, score, frame, overAt = 0, best = 0;
let health = 100, enemies = [], attacks = [], particles = [];
let lastEnemySpawn = 0, nextAttackAt = 0, parryUntil = 0, parryReadyAt = 0;
let parryFlashUntil = 0, parryStartedAt = 0, message = "", messageUntil = 0;
const MAX_ENEMIES = 6, ENEMY_SPAWN_MS = 4200, ATTACK_INTERVAL_MS = 6000;
const SPECIES = [
  { name:"Razor", kind:"melee", color:"#C44747", accent:"#7D2424" },
  { name:"Gunner", kind:"ranged", color:"#4776B8", accent:"#24466F" },
  { name:"Stinger", kind:"melee", color:"#8A58A8", accent:"#51316A" },
  { name:"Bomber", kind:"ranged", color:"#398C72", accent:"#205C49" }
];
try { best = Number(localStorage.getItem("flappyAiJamBest")) || 0; } catch (e) {}

function resize() {
  const chrome = document.querySelector("header").offsetHeight + document.querySelector("footer").offsetHeight + 24;
  const scale = Math.max(0.2, Math.min((innerWidth - 24) / W, (innerHeight - chrome) / H));
  canvas.style.width = Math.floor(W * scale) + "px";
  canvas.style.height = Math.floor(H * scale) + "px";
}
addEventListener("resize", resize);
resize();

function reset() {
  bird = { x:100, y:H / 2, vy:0, r:16 };
  pipes = []; score = 0; frame = 0; health = 100;
  enemies = []; attacks = []; particles = [];
  lastEnemySpawn = performance.now(); nextAttackAt = performance.now() + ATTACK_INTERVAL_MS;
  parryUntil = 0; parryReadyAt = 0; parryFlashUntil = 0; parryStartedAt = 0;
  message = ""; messageUntil = 0;
}
function flap() {
  if (state === "over") {
    if (performance.now() - overAt < 300) return;
    reset();
  }
  state = "playing";
  bird.vy = FLAP_STRENGTH;
}
function tryParry() {
  if (state !== "playing") return;
  const now = performance.now();
  if (now < parryReadyAt) {
    message = "PARRY RECHARGING"; messageUntil = now + 500;
    return;
  }
  parryUntil = now + 1000; // 1-second parry window
  parryFlashUntil = now + 140;
}
function updateBird() {
  bird.vy += GRAVITY;
  bird.y += bird.vy;
  if (bird.y - bird.r < 0) { bird.y = bird.r; bird.vy = 0; }
  if (bird.y + bird.r > H - GROUND_H) { bird.y = H - GROUND_H - bird.r; bird.vy = 0; }
}
function updatePipes() {
  const last = pipes[pipes.length - 1];
  if (!last || last.x < W - PIPE_SPACING) {
    pipes.push({ x:W, top:60 + Math.random() * (H - GROUND_H - PIPE_GAP - 120), passed:false });
  }
  for (const p of pipes) {
    p.x -= PIPE_SPEED;
    const hitX = bird.x + bird.r > p.x && bird.x - bird.r < p.x + PIPE_WIDTH;
    const hitY = bird.y - bird.r < p.top || bird.y + bird.r > p.top + PIPE_GAP;
    if (hitX && hitY && !p.hit) { p.hit = true; damagePlayer(); }
    if (!p.passed && p.x + PIPE_WIDTH < bird.x) { p.passed = true; score++; }
  }
  pipes = pipes.filter(p => p.x + PIPE_WIDTH > -10);
}
function spawnEnemy(now) {
  // Spawn one coordinated six-enemy wave once the player reaches 10 points.
  if (score < 10 || enemies.length > 0 || now - lastEnemySpawn < ENEMY_SPAWN_MS) return;
  const topXs = [185, 255, 325];
  const bottomXs = [185, 255, 325];
  for (let i = 0; i < 6; i++) {
    const species = SPECIES[Math.floor(Math.random() * SPECIES.length)];
    const fromTop = i < 3;
    const targetX = (fromTop ? topXs[i] : bottomXs[i - 3]);
    const targetY = fromTop ? 95 + Math.random() * 55 : 390 + Math.random() * 55;
    enemies.push({
      x: targetX + (Math.random() - 0.5) * 30,
      y: fromTop ? -40 - i * 12 : H + 40 + (i - 3) * 12,
      targetX, targetY, species, hp:3, maxHp:3,
      born:now, alpha:1, phase:"enter", attackAt:0, fadeAt:0,
      seed:Math.random() * Math.PI * 2
    });
  }
  lastEnemySpawn = now;
}
function activeAttackCount() {
  const now = performance.now();
  return attacks.filter(a => !a.done && now < a.impactAt + 250).length;
}
function launchAttacks(now) {
  if (score < 10 || now < nextAttackAt) return;
  const candidates = enemies.filter(e => e.phase === "ready" && !e.attackAt);
  if (!candidates.length) {
    nextAttackAt = now + 500;
    return;
  }
  const enemy = candidates[Math.floor(Math.random() * candidates.length)];
  enemy.attackAt = now;
  const ranged = enemy.species.kind === "ranged";
  const windup = ranged ? 1000 : 500;
  const travel = ranged ? 500 : 1000;
  attacks.push({
    enemy, kind:enemy.species.kind, startAt:now,
    windupAt:now + windup, impactAt:now + windup + travel,
    targetX:bird.x, targetY:bird.y,
    done:false, reflected:false, fromX:enemy.x, fromY:enemy.y
  });
  nextAttackAt = now + ATTACK_INTERVAL_MS;
}
function damagePlayer() {
  if (state !== "playing") return;
  health = Math.max(0, health - 20);
  message = "-20 HEALTH"; messageUntil = performance.now() + 650;
  if (health <= 0) gameOver();
}
function hitEnemy(enemy) {
  if (!enemy || enemy.phase === "fading") return;
  enemy.hp--;
  for (let i=0;i<8;i++) particles.push({x:enemy.x,y:enemy.y,vx:(Math.random()-.5)*4,vy:(Math.random()-.5)*4,life:24});
  if (enemy.hp <= 0) { enemy.phase = "fading"; enemy.fadeAt = performance.now(); }
  else { message = "PARRY HIT!"; messageUntil = performance.now() + 500; }
}
function getAttackPosition(a, now) {
  const duration = Math.max(1, a.impactAt - a.windupAt);
  const t = Math.max(0, Math.min(1, (now - a.windupAt) / duration));
  return {
    x: a.fromX + (a.targetX - a.fromX) * t,
    y: a.fromY + (a.targetY - a.fromY) * t
  };
}
function updateCombat(now) {
  spawnEnemy(now);
  for (const e of enemies) {
    if (e.phase === "enter") {
      e.x += (e.targetX - e.x) * 0.035;
      e.y += (e.targetY - e.y) * 0.035;
      if (Math.abs(e.x-e.targetX) < 5 && Math.abs(e.y-e.targetY) < 5) e.phase = "ready";
    } else if (e.phase === "ready") {
      e.y += Math.sin(now/420 + e.seed) * 0.35;
    } else if (e.phase === "fading") {
      e.alpha = Math.max(0, 1 - (now-e.fadeAt)/500);
    }
  }
  launchAttacks(now);
  for (const a of attacks) {
    if (a.done || a.reflected || now < a.windupAt) continue;
    const pos = getAttackPosition(a, now);
    const contactRadius = a.kind === "ranged" ? bird.r + 7 : bird.r + 13;
    if (Math.hypot(bird.x - pos.x, bird.y - pos.y) <= contactRadius) {
      a.done = true;
      damagePlayer();
    } else if (now >= a.impactAt) {
      // The projectile has completed its travel and missed the bird.
      a.done = true;
    }
  }
  attacks = attacks.filter(a => !a.done || now < a.impactAt + 180);
  enemies = enemies.filter(e => e.phase !== "fading" || e.alpha > 0);
  particles = particles.filter(p => p.life-- > 0);
  for (const p of particles) { p.x += p.vx; p.y += p.vy; p.vy += .12; }
  if (now > parryUntil && parryUntil && parryUntil < now) {
    // A parry attempt with no attack crossing its active window is a failed parry.
    if (!parryUntil._checked) {}
  }
}
// Check parry against an attack during its exact 50ms active window.
function resolveParry(now) {
  if (!parryUntil || now < parryStartedAt || now > parryUntil) return;
  for (const a of attacks) {
    if (a.done || a.reflected || now < a.windupAt || now > a.impactAt) continue;
    const pos = getAttackPosition(a, now);
    const contactRadius = a.kind === "ranged" ? bird.r + 7 : bird.r + 13;
    if (Math.hypot(bird.x - pos.x, bird.y - pos.y) <= contactRadius) {
      a.reflected = true;
      a.done = true;
      hitEnemy(a.enemy);
      parryReadyAt = now;
      parryUntil = 0;
      parryAttemptPending = false;
      message = "PERFECT PARRY!";
      messageUntil = now + 700;
      return;
    }
  }
}
let parryAttemptPending = false;
function updateParry(now) {
  if (parryAttemptPending && parryUntil && now > parryUntil) {
    const wasSuccessful = parryReadyAt === now || parryReadyAt > parryUntil - 100;
    if (!wasSuccessful) {
      parryReadyAt = now + 4500;
      message = "PARRY MISSED"; messageUntil = now + 650;
    }
    parryAttemptPending = false;
    parryUntil = 0;
  }
}
function drawBackground() {
  ctx.fillStyle = COLORS.cream; ctx.fillRect(0, 0, W, H);
  ctx.fillStyle = "rgba(7,59,92,0.08)";
  for (let i = 0; i < 4; i++) {
    const cx = ((i * 140 - frame * 0.4) % (W + 120) + W + 120) % (W + 120) - 60;
    ctx.beginPath(); ctx.arc(cx, 90 + i * 45, 30, 0, Math.PI * 2); ctx.arc(cx + 30, 85 + i * 45, 22, 0, Math.PI * 2); ctx.fill();
  }
}
function drawPipes() {
  for (const p of pipes) {
    ctx.fillStyle = COLORS.pipe;
    ctx.fillRect(p.x, 0, PIPE_WIDTH, p.top);
    ctx.fillRect(p.x, p.top + PIPE_GAP, PIPE_WIDTH, H - GROUND_H - p.top - PIPE_GAP);
    ctx.fillStyle = COLORS.navy;
    ctx.fillRect(p.x - 4, p.top - 20, PIPE_WIDTH + 8, 20);
    ctx.fillRect(p.x - 4, p.top + PIPE_GAP, PIPE_WIDTH + 8, 20);
  }
}
function drawGround() {
  ctx.fillStyle = COLORS.ground; ctx.fillRect(0, H - GROUND_H, W, GROUND_H);
  ctx.fillStyle = COLORS.navy; ctx.fillRect(0, H - GROUND_H, W, 4);
}
function drawBird() {
  ctx.save(); ctx.translate(bird.x, bird.y);
  ctx.rotate(Math.max(-0.5, Math.min(1.2, bird.vy * 0.08)));
  ctx.fillStyle = COLORS.bird; ctx.beginPath(); ctx.arc(0, 0, bird.r, 0, Math.PI * 2); ctx.fill();
  // Feather/wing used for the parry animation.
  ctx.save(); ctx.rotate(parryFlashUntil > performance.now() ? -0.9 : 0.15);
  ctx.fillStyle = "#F2A900"; ctx.beginPath(); ctx.ellipse(-5, 5, 10, 4, -0.4, 0, Math.PI*2); ctx.fill(); ctx.restore();
  ctx.fillStyle = "#fff"; ctx.beginPath(); ctx.arc(6, -5, 6, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = COLORS.navy; ctx.beginPath(); ctx.arc(8, -5, 3, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = COLORS.beak; ctx.beginPath(); ctx.moveTo(12, 0); ctx.lineTo(24, 4); ctx.lineTo(12, 8); ctx.fill();
  ctx.restore();
}
function drawEnemies(now) {
  for (const e of enemies) {
    const s = e.species;
    ctx.save(); ctx.globalAlpha = e.alpha;
    ctx.translate(e.x,e.y);
    ctx.fillStyle = s.color; ctx.beginPath(); ctx.ellipse(0,0,20,13,0,0,Math.PI*2); ctx.fill();
    ctx.fillStyle = s.accent;
    ctx.beginPath(); ctx.moveTo(-10,-7); ctx.lineTo(-22,-20); ctx.lineTo(-17,-3); ctx.fill();
    ctx.beginPath(); ctx.moveTo(-10,7); ctx.lineTo(-22,20); ctx.lineTo(-17,3); ctx.fill();
    ctx.fillStyle = "#fff"; ctx.beginPath(); ctx.arc(9,-4,4,0,Math.PI*2); ctx.fill();
    ctx.fillStyle = "#172B3A"; ctx.beginPath(); ctx.arc(10,-4,2,0,Math.PI*2); ctx.fill();
    if (s.kind === "ranged") {
      ctx.fillStyle = "#333"; ctx.fillRect(9,5,20,7); ctx.fillRect(23,3,7,11);
    } else {
      ctx.strokeStyle = "#D8DCE2"; ctx.lineWidth = 3; ctx.beginPath(); ctx.moveTo(12,8); ctx.lineTo(27,20); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(24,15); ctx.lineTo(30,22); ctx.stroke();
    }
    ctx.fillStyle = "#263746"; ctx.fillRect(-18,-25,36,5);
    ctx.fillStyle = "#67B96B"; ctx.fillRect(-18,-25,36*(e.hp/3),5);
    ctx.restore();
  }
}
function drawAttacks(now) {
  for (const a of attacks) {
    if (a.done && !a.reflected) continue;
    const e = a.enemy;
    if (a.reflected) {
      ctx.strokeStyle = "#F2A900"; ctx.lineWidth = 4;
      ctx.beginPath(); ctx.moveTo(bird.x+10,bird.y); ctx.lineTo(e.x,e.y); ctx.stroke();
      continue;
    }
    if (now < a.startAt) continue;
    if (now < a.windupAt) {
      ctx.strokeStyle = a.kind === "ranged" ? "#B56B27" : "#9C3C3C";
      ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(e.x,e.y,25,0,Math.PI*2); ctx.stroke();
      ctx.fillStyle = COLORS.navy; ctx.font = "10px system-ui"; ctx.textAlign = "center";
      ctx.fillText(a.kind === "ranged" ? "AIMING" : "SLASH!",e.x,e.y-31);
    } else if (now <= a.impactAt) {
      const pos = getAttackPosition(a, now);
      const x = pos.x, y = pos.y;
      if (a.kind === "ranged") {
        ctx.fillStyle = "#333"; ctx.beginPath(); ctx.arc(x,y,6,0,Math.PI*2); ctx.fill();
        ctx.fillStyle = "#E28A35"; ctx.beginPath(); ctx.arc(x-2,y-2,2,0,Math.PI*2); ctx.fill();
      } else {
        ctx.strokeStyle = "#B7C2CC"; ctx.lineWidth = 6; ctx.beginPath(); ctx.moveTo(x-12,y+12); ctx.lineTo(x+12,y-12); ctx.stroke();
      }
    }
  }
  for (const p of particles) { ctx.fillStyle = "#F2A900"; ctx.fillRect(p.x,p.y,4,4); }
}
function drawText(text, y, size) {
  ctx.fillStyle = COLORS.navy; ctx.font = "bold " + size + "px system-ui, sans-serif";
  ctx.textAlign = "center"; ctx.fillText(text, W / 2, y);
}
function drawBars() {
  const x=20, w=150, h=12;
  // Parry bar: bright when ready, dull while the 4.5s lockout is active.
  const now=performance.now(), remaining=Math.max(0,parryReadyAt-now);
  ctx.fillStyle="#25394A"; ctx.fillRect(x,18,w,h);
  ctx.fillStyle=remaining>0?"#8B9298":"#36A6A0"; ctx.fillRect(x,18,w*(remaining>0?remaining/4500:1),h);
  ctx.strokeStyle=COLORS.navy; ctx.strokeRect(x,18,w,h);
  ctx.fillStyle=COLORS.navy; ctx.font="bold 10px system-ui"; ctx.textAlign="left";
  ctx.fillText(remaining>0?"PARRY  "+(remaining/1000).toFixed(1)+"s":"PARRY READY",x,13);
  ctx.fillStyle="#25394A"; ctx.fillRect(x,45,w,h);
  ctx.fillStyle="#D65B54"; ctx.fillRect(x,45,w*(health/100),h);
  ctx.strokeStyle=COLORS.navy; ctx.strokeRect(x,45,w,h);
  ctx.fillStyle=COLORS.navy; ctx.fillText("HEALTH  "+health+"/100",x,40);
}
function drawUI() {
  if (state === "playing") {
    drawText(score, 95, 48); drawBars();
    ctx.fillStyle=COLORS.navy; ctx.font="11px system-ui"; ctx.textAlign="right";
    ctx.fillText("ENEMIES "+enemies.length+"/6",W-16,24);
    if (message && performance.now()<messageUntil) drawText(message,H-95,16);
  }
  if (state === "start") {
    drawText("FLAPPY COMBAT", H / 2 - 105, 34);
    drawText("SPACE / CLICK: FLY", H / 2 - 45, 17);
    drawText("F: PARRY ATTACKS", H / 2 - 15, 17);
    drawText("Reach 10 points to face predators", H / 2 + 35, 14);
    drawText("Best: " + best, H / 2 + 75, 18);
  }
  if (state === "over") {
    ctx.fillStyle = COLORS.cream; ctx.fillRect(40,H/2-110,W-80,200);
    ctx.strokeStyle=COLORS.navy; ctx.lineWidth=4; ctx.strokeRect(40,H/2-110,W-80,200);
    drawText("GAME OVER",H/2-60,36);
    drawText("Score: "+score+"   Best: "+best,H/2-10,20);
    drawText("Tap / Space to restart",H/2+40,16);
  }
}
function gameOver() {
  if (state !== "playing") return;
  state = "over"; overAt = performance.now();
  if (score > best) { best=score; try { localStorage.setItem("flappyAiJamBest",best); } catch(e) {} }
}
function loop() {
  frame++;
  const now=performance.now();
  const label=state==="playing"?"Playing":state==="over"?"Game over":"Ready to play";
  if(statusEl.textContent!==label) statusEl.textContent=label;
  if(state==="playing") {
    updateBird(); updatePipes();
    if(state==="playing") { resolveParry(now); updateCombat(now); updateParry(now); }
  } else if(state==="start") bird.y=H/2+Math.sin(frame/15)*10;
  drawBackground(); drawPipes(); drawGround(); drawEnemies(now); drawAttacks(now); drawBird(); drawUI();
  requestAnimationFrame(loop);
}
addEventListener("keydown", e => {
  if(e.defaultPrevented) return;
  if(e.code==="Space"||e.code==="ArrowUp") { e.preventDefault(); flap(); }
  if(e.code==="KeyF"&&!e.repeat) {
    e.preventDefault();
    if(state==="playing" && performance.now()>=parryReadyAt) {
      parryStartedAt=performance.now(); parryUntil=parryStartedAt+1000; parryAttemptPending=true; parryReadyAt=0;
    }
  }
});
canvas.addEventListener("click",flap);
reset();
