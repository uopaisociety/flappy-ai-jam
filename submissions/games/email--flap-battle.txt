// === AI PROMPT ZONE: CONFIG ===
const W = 400, H = 600;
const PIPE_GAP = 205, MIN_PIPE_GAP = 145, PIPE_WIDTH = 58, PIPE_SPACING = 260;
const PIPE_SPEED = 140, MAX_PIPE_SPEED = 220, RAMP_DELAY = 8, RAMP_SECONDS = 60;
const GROUND_H = 50, STEP = 1 / 120, START_Y = 285;
const GRAVITY = 880, FLAP_STRENGTH = -360, MAX_FALL_SPEED = 400;
const FLICK_ANGLE = 16 * Math.PI / 180, FLICK_SPEED = 2.6, FLICK_COOLDOWN = 145;
const COLORS = { navy:"#073B5C", cream:"#F7F5F0", bird:"#FFC93C", beak:"#FF7A00", pipe:"#0E5A8A", ground:"#C9B79C" };
const $ = id => document.getElementById(id);
const ctx = $("game").getContext("2d");
const clamp = (value, low, high) => Math.min(high, Math.max(low, value));
const players = ["blue", "pink"].map((team, i) => ({
  id:i + 1, team, color:i ? "#aa2865" : "#096aa9",
  up:i ? "ArrowUp" : "KeyW", down:i ? "ArrowDown" : "KeyS", hand:null
}));
const keys = new Set();
let state = "menu", pausedState, mode = "keyboard";
let pipes, elapsed, nextCenter, countdown, accumulator = 0, lastTime = null;
let winner, confetti = [];
let stream, detectorPromise, cameraSession = 0, cameraReady = false;
const video = $("video");

function resetRace() {
  pipes = [];
  elapsed = 0;
  nextCenter = START_Y;
  countdown = 3;
  for (const player of players) {
    Object.assign(player, { x:100, y:START_Y, vy:0, r:15, alive:true, score:0, flap:false });
  }
}

function showScreen(screen) {
  state = screen;
  const showCamera = screen === "setup" || (mode === "camera" && screen !== "menu");
  $("liveCamera").hidden = !showCamera;
  document.body.classList.toggle("camera-on", showCamera);
  for (const id of ["menu", "setup", "pause", "result"]) $(id).hidden = id !== screen;
  $("pauseButton").hidden = screen !== "countdown" && screen !== "playing";
  $("count").hidden = screen !== "countdown";
  accumulator = 0;
  lastTime = null;
}

function startRace() {
  resetRace();
  $("count").textContent = "3";
  showScreen("countdown");
  updateHUD();
}

function updateHUD(now = performance.now()) {
  for (const player of players) {
    let label = "Player " + player.id;
    if (!player.alive) label += " · Out";
    else if (mode === "camera" && state === "playing" && !isTracked(player, now)) label += " · Keys";
    setText($("p" + player.id), label);
    setText($("score" + player.id), player.score);
  }
}

// === AI PROMPT ZONE: PHYSICS & INPUT ===
function updateBird(player, dt) {
  if (player.flap) {
    player.vy = FLAP_STRENGTH;
    player.flap = false;
  }
  if (keys.has(player.down)) player.vy = Math.max(player.vy, 180);
  player.vy = Math.min(MAX_FALL_SPEED, player.vy + GRAVITY * dt);
  player.y += player.vy * dt;
  if (player.y < player.r) {
    player.y = player.r;
    player.vy = 0;
  }
  if (player.y + player.r > H - GROUND_H) {
    player.y = H - GROUND_H - player.r;
    crash(player);
  }
}

// === AI PROMPT ZONE: PIPES / LEVEL ===
function updatePipes(dt) {
  const difficulty = clamp((elapsed - RAMP_DELAY) / RAMP_SECONDS, 0, 1);
  const speed = PIPE_SPEED + (MAX_PIPE_SPEED - PIPE_SPEED) * difficulty;
  const last = pipes[pipes.length - 1];
  if (!last || last.x <= W - PIPE_SPACING) {
    const gap = PIPE_GAP + (MIN_PIPE_GAP - PIPE_GAP) * difficulty;
    const drift = 55 + 30 * difficulty;
    if (players.some(player => player.score > 0)) {
      nextCenter = clamp(nextCenter + (Math.random() * 2 - 1) * drift,
        gap / 2 + 50, H - GROUND_H - gap / 2 - 50);
    }
    pipes.push({ x:last ? W : W + 100, top:nextCenter - gap / 2, gap, passed:false });
  }
  for (const pipe of pipes) {
    pipe.x -= speed * dt;
    for (const player of players) {
      if (!player.alive) continue;
      const overlaps = player.x + player.r - 3 > pipe.x && player.x - player.r + 3 < pipe.x + PIPE_WIDTH;
      const outsideGap = player.y - player.r + 3 < pipe.top || player.y + player.r - 3 > pipe.top + pipe.gap;
      if (overlaps && outsideGap) crash(player);
    }
    if (!pipe.passed && pipe.x + PIPE_WIDTH < players[0].x - players[0].r) {
      pipe.passed = true;
      for (const player of players) if (player.alive) player.score++;
    }
  }
  pipes = pipes.filter(pipe => pipe.x + PIPE_WIDTH > 0);
}

// === AI PROMPT ZONE: SCORE & GAME OVER ===
function crash(player) {
  player.alive = false;
  player.vy = 230;
}

function finishRace() {
  const [left, right] = players;
  winner = left.score === right.score ? null : left.score > right.score ? left : right;
  const title = $("resultTitle");
  title.className = winner?.team || "";
  title.textContent = winner ? "Player " + winner.id + " wins!" : "Race tied!";
  $("resultBody").textContent = "Player 1: " + left.score + " · Player 2: " + right.score + " pipes";
  showScreen("result");
  $("next").focus();
  if (winner && !matchMedia("(prefers-reduced-motion: reduce)").matches) {
    title.animate([{transform:"scale(.85)"},{transform:"scale(1.06)"},{transform:"scale(1)"}], 360);
    const start = performance.now();
    confetti = Array.from({length:70}, () => ({
      x:Math.random() * 800, y:-Math.random() * 230, speed:190 + Math.random() * 260,
      spin:Math.random() * 8, color:Math.random() < .65 ? winner.color : COLORS.bird, start
    }));
  }
}

// === AI PROMPT ZONE: GRAPHICS ===
// Draw with shapes. Keep controls and the score legible over any new artwork.
function drawBackground() {
  ctx.fillStyle = COLORS.cream;
  ctx.fillRect(0, 0, W, H);
  ctx.fillStyle = "rgba(7,59,92,0.08)";
  for (let i = 0; i < 4; i++) {
    const x = ((i * 140 - elapsed * 24) % (W + 120) + W + 120) % (W + 120) - 60;
    ctx.beginPath();
    ctx.arc(x, 80 + i * 45, 30, 0, Math.PI * 2);
    ctx.arc(x + 30, 75 + i * 45, 22, 0, Math.PI * 2);
    ctx.fill();
  }
}
function drawPipes() {
  for (const pipe of pipes) {
    ctx.fillStyle = COLORS.pipe;
    ctx.fillRect(pipe.x, 0, PIPE_WIDTH, pipe.top);
    ctx.fillRect(pipe.x, pipe.top + pipe.gap, PIPE_WIDTH, H - GROUND_H - pipe.top - pipe.gap);
    ctx.fillStyle = COLORS.navy;
    ctx.fillRect(pipe.x, pipe.top - 16, PIPE_WIDTH, 16);
    ctx.fillRect(pipe.x, pipe.top + pipe.gap, PIPE_WIDTH, 16);
  }
}
function drawBird(bird, color, opacity = 1) {
  ctx.save();
  ctx.translate(bird.x, bird.y);
  ctx.rotate(Math.max(-0.4, Math.min(0.8, bird.vy / 500)));
  ctx.globalAlpha = opacity * (bird.alive ? 1 : 0.45);
  if (bird.alive) {
    ctx.strokeStyle = color;
    ctx.lineWidth = 2;
    ctx.beginPath(); ctx.arc(0, 0, bird.r + 7, 0, Math.PI * 2); ctx.stroke();
  }
  ctx.fillStyle = COLORS.bird;
  ctx.beginPath(); ctx.arc(0, 0, bird.r, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = "#fff";
  ctx.beginPath(); ctx.arc(6, -5, 6, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = COLORS.navy;
  ctx.beginPath(); ctx.arc(8, -5, 3, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = COLORS.beak;
  ctx.beginPath(); ctx.moveTo(12, 0); ctx.lineTo(24, 4); ctx.lineTo(12, 8); ctx.fill();
  ctx.restore();
}
function drawText(text, y, size) {
  ctx.fillStyle = COLORS.navy;
  ctx.font = "bold " + size + "px system-ui, sans-serif";
  ctx.textAlign = "center";
  ctx.fillText(String(text), W / 2, y);
}


function draw(now) {
  for (const [i, player] of players.entries()) {
    ctx.save();
    ctx.translate(i * W, 0);
    ctx.beginPath();
    ctx.rect(0, 0, W, H);
    ctx.clip();
    drawBackground();
    drawPipes();
    const rival = players[1 - i];
    drawBird(rival, rival.color, .33);
    drawBird(player, player.color);
    ctx.fillStyle = COLORS.ground;
    ctx.fillRect(0, H - GROUND_H, W, GROUND_H);
    if (!player.alive) drawText("OUT", H - 17, 21);
    ctx.restore();
  }
  ctx.fillStyle = COLORS.navy;
  ctx.fillRect(W - 2, 0, 4, H);
  if (state !== "result") return;
  for (const bit of confetti) {
    const time = (now - bit.start) / 1000;
    const y = bit.y + bit.speed * time;
    if (y > H) continue;
    ctx.save();
    ctx.translate(bit.x + Math.sin(time * 4 + bit.spin) * 18, y);
    ctx.rotate(time * bit.spin);
    ctx.fillStyle = bit.color;
    ctx.fillRect(-4, -3, 8, 6);
    ctx.restore();
  }
}

      return HandLandmarker.createFromOptions(files, options);
    }
  })().catch(error => { detectorPromise = null; throw error; });
  return detectorPromise;
}

function stopCamera() {
  cameraSession++;
  stream?.getTracks().forEach(track => track.stop());
  stream = null;
  video.srcObject = null;
  cameraReady = false;
  players.forEach(player => { player.hand = null; });
}

async function setupCamera() {
  stopCamera();
  showScreen("setup");
  updateCameraPreview(performance.now());
  $("cameraMessage").textContent = "Allow camera access to get started.";
  $("retry").hidden = true;
  const session = cameraSession;
  try {
    if (!navigator.mediaDevices?.getUserMedia) throw new Error("secure-context");
    const input = await navigator.mediaDevices.getUserMedia({
      video:{ width:{ideal:480}, height:{ideal:360}, frameRate:{ideal:30, max:30} }, audio:false
    });
    if (session !== cameraSession) {
      input.getTracks().forEach(track => track.stop());
      return;
    }
    stream = input;
    input.getVideoTracks()[0].addEventListener("ended", () => {
      if (session === cameraSession) cameraError("Camera disconnected.");
    });
    video.srcObject = input;
    await video.play();
    if (session !== cameraSession) return;
    $("cameraMessage").textContent = "Loading hand tracking…";
    detector = await loadDetector();
    if (session !== cameraSession) return;
    previousFrame = -1;
    cameraReady = true;
    $("cameraMessage").textContent = "Use the preview below. P1 on the left, P2 on the right. Keep your wrist in place and flick up or down.";
  } catch (error) {
    if (session !== cameraSession) return;
    const message = error.message === "secure-context" ? "Open the game on localhost or HTTPS to use the camera." :
      error.name === "NotAllowedError" ? "Camera access was blocked. Check browser permissions." :
      error.name === "NotFoundError" ? "No webcam was found." : "Camera setup failed. Check your connection or use the keyboard.";
    cameraError(message);
  }
}

function cameraError(message) {
  stopCamera();
  if (state === "setup") {
    updateCameraPreview(performance.now());
    $("cameraMessage").textContent = message;
    $("retry").hidden = false;
  }
}

function isTracked(player, now) {
  return !!player.hand && now - player.hand.seen < 220;
}

function normalize(vector) {
  const length = Math.hypot(...vector);
  return length > 1e-6 ? vector.map(value => value / length) : null;
}

function palmDirection(points, aspect) {
  const wrist = points[0];
  const knuckles = [5, 9, 13, 17].map(index => points[index]);
  return normalize(["x", "y", "z"].map(axis => {
    const displacement = knuckles.reduce((sum, point) => sum + point[axis] - wrist[axis], 0) / knuckles.length;
    return axis === "y" ? displacement * aspect : displacement;
  }));
}

function assignHands(result, now) {
  const hands = result.landmarks.map((points, i) => {
    const world = result.worldLandmarks?.[i];
    const aspect = video.videoWidth ? video.videoHeight / video.videoWidth : 1;
    const direction = world && palmDirection(world, 1) || palmDirection(points, aspect);
    return { x:1 - points[0].x, y:points[0].y, direction };
  });

  for (const player of players) {
    const previous = isTracked(player, now) ? player.hand : null;
    // Each wrist stays in its own preview half, even when only one hand is visible.
    const candidates = hands.filter(hand => (hand.x < .5) === (player.id === 1));
    const anchor = previous?.x ?? (player.id === 1 ? .25 : .75);
    const hand = candidates.sort((a, b) => Math.abs(a.x - anchor) - Math.abs(b.x - anchor))[0];
    if (!hand) continue;
    hand.seen = now;
    hand.lastFlick = previous?.lastFlick ?? -Infinity;
    hand.samples = previous?.samples.filter(sample => now - sample.time <= 140) ?? [];
    if (hand.direction) {
      // Compare wrist-to-knuckle directions so translating the whole hand cannot flap.
      const flick = now - hand.lastFlick >= FLICK_COOLDOWN && hand.samples.some(sample => {
        const seconds = (now - sample.time) / 1000;
        if (seconds < .025) return false;
        const dot = sample.direction.reduce((sum, value, i) => sum + value * hand.direction[i], 0);
        const angle = Math.acos(clamp(dot, -1, 1));
        return angle >= FLICK_ANGLE && angle >= FLICK_SPEED * seconds;
      });
      if (flick) {
        hand.lastFlick = now;
        hand.samples = [];
        if (state === "playing" && player.alive) player.flap = true;
      }
      hand.samples.push({ direction:hand.direction, time:now });
    }
    player.hand = hand;
  }
}

function pollCamera(now) {
  if (!cameraReady || video.readyState < 2 || now - lastInference < 32 || previousFrame === video.currentTime) return;
  previousFrame = video.currentTime;
  lastInference = now;
  try {
    assignHands(detector.detectForVideo(video, now), now);
  } catch {
    detector?.close();
    detector = null;
    detectorPromise = null;
    cameraError("Hand tracking stopped. Retry the camera or use the keyboard.");
  }
}

function updateCameraPreview(now) {
  for (const player of players) {
    const ready = cameraReady && isTracked(player, now);
    const marker = $("marker" + player.id);
    marker.hidden = !ready;
    if (ready) {
      marker.style.left = (player.hand.x * 100) + "%";
      marker.style.top = (player.hand.y * 100) + "%";
    }
    const flicked = ready && now - player.hand.lastFlick < 400;
    setText($("hand" + player.id), "P" + player.id + (flicked ? " · Flap!" : ready ? " · Ready" : " · Searching"));
  }
  $("cameraPlay").disabled = !cameraReady || !players.every(player => isTracked(player, now));
}

// === AI PROMPT ZONE: GAME LOOP ===
function loop(now) {
  if (!document.hidden) pollCamera(now);
  if (!$("liveCamera").hidden) updateCameraPreview(now);
  if (lastTime !== null && (state === "countdown" || state === "playing")) {
    const dt = Math.min(.1, Math.max(0, (now - lastTime) / 1000));
    if (state === "countdown") {
      countdown -= dt;
      setText($("count"), Math.max(1, Math.ceil(countdown)));
      if (countdown <= 0) {
        players.forEach(player => { player.flap = true; });
        showScreen("playing");
      }
    } else {
      accumulator += dt;
      while (accumulator >= STEP && state === "playing") {
        accumulator -= STEP;
        elapsed += STEP;
        for (const player of players) if (player.alive) updateBird(player, STEP);
        updatePipes(STEP);
        if (players.every(player => !player.alive)) finishRace();
      }
      updateHUD(now);
    }
  }
  lastTime = now;
  draw(now);
  requestAnimationFrame(loop);
}
