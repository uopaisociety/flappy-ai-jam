<script>
"use strict";

// === AI PROMPT ZONE: CONFIG ===
// Positions are pixels, speed is pixels/second, gravity is pixels/second².
// Start with colours or one mechanic. The first gates teach the rhythm. Speed and gap difficulty then ramp over a minute.
const W = 400, H = 600;
const GRAVITY = 880, FLAP_STRENGTH = -285, MAX_FALL_SPEED = 400;
const PIPE_GAP = 205, MIN_PIPE_GAP = 145, PIPE_WIDTH = 58, PIPE_SPACING = 260;
const PIPE_SPEED = 140, MAX_PIPE_SPEED = 220, RAMP_DELAY = 8, RAMP_SECONDS = 60;
const GROUND_H = 50, START_LIVES = 3, RECOVERY_SECONDS = 1.2, STEP = 1 / 120;
// === AI PROMPT ZONE: FEATURES ===
// Switch each add-on idea on or off. The core game works with all of them off.
const FEATURES = { gold:true, hearts:true, night:true, sound:true, handCam:true, streak:true, demo:true };
const TAGLINE = FEATURES.gold && FEATURES.hearts ? "Gold chips +5. Hearts give a life."
  : FEATURES.gold ? "Gold chips are worth +5." : FEATURES.hearts ? "Hearts give a life." : "Chips are worth +1.";

const CHIP_CHANCE = 0.7;   // Chance that a gap holds a chip worth one bonus point.
const GOLD_CHANCE = 0.1;   // Share of gaps holding a golden chip worth GOLD_POINTS.
const HEART_CHANCE = 0.05; // Share of gaps holding a bonus life (from score 3, below MAX_LIVES).
const GOLD_POINTS = 5, MAX_LIVES = 5, NIGHT_SCORE = 30, STREAK_STEP = 5;
const COLORS = { navy:"#073B5C", cream:"#F7F5F0", skyTop:"#FF9E6B", skyLow:"#FDE6C4", sun:"#FFD36E",
  sea:"#2F7FA8", seaDark:"#1D5F82", gull:"#FFFFFF", wing:"#AEB9C4", tip:"#1E2A33", beak:"#FFC23C",
  stripe:"#D7263D", pipe:"#F7F5F0", ground:"#D9C7A7", pebble:"#B39D7C", chip:"#F2C14E", cone:"#C8102E", tower:"rgba(7,59,92,0.22)" };
const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");
const statusEl = document.getElementById("status");
let state = "start", bird, pipes, score, chips, lives, recovery, elapsed, nextCenter, night, dusk, popups;
let streak = 0, bestStreak = 0, callout = null;
let audio = null, muted = false;
let overAt = 0, best = 0, lastTime = null, accumulator = 0;
try {
  const stored = Number(localStorage.getItem("flappyAiJamBest-seagull-snatch-deluxe"));
  if (Number.isSafeInteger(stored) && stored > 0) best = stored;
} catch { /* The game also works when browser storage is blocked. */ }

const playArea = canvas.parentElement;

function resize() {
  const scale = Math.min((playArea.clientWidth - 24) / W, (playArea.clientHeight - 24) / H);
  if (scale <= 0) return;
  canvas.style.width = Math.floor(W * scale) + "px";
  canvas.style.height = Math.floor(H * scale) + "px";
}
// The play area can gain height after the iframe lays out or the controls reflow.
if (typeof ResizeObserver === "function") new ResizeObserver(resize).observe(playArea);
else addEventListener("resize", resize);
resize();

function reset() {
  bird = { x:100, y:285, vy:0, r:15 };
  pipes = [];
  score = chips = elapsed = recovery = night = dusk = streak = bestStreak = 0;
  callout = null;
  assisted = false;
  popups = [];
  lives = START_LIVES;
  nextCenter = bird.y;
}

// === AI PROMPT ZONE: SOUND ===
// Sounds are made with Web Audio oscillators, so there are no files. Any failure stays silent.
function tone(type, from, to, length, volume) {
  if (muted || !FEATURES.sound) return;
  try {
    audio = audio || new (window.AudioContext || window.webkitAudioContext)();
    if (audio.state === "suspended") audio.resume();
    const t = audio.currentTime, osc = audio.createOscillator(), gain = audio.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(from, t);
    osc.frequency.exponentialRampToValueAtTime(to, t + length);
    gain.gain.setValueAtTime(volume, t);
    gain.gain.exponentialRampToValueAtTime(0.001, t + length);
    osc.connect(gain).connect(audio.destination);
    osc.start(t); osc.stop(t + length);
  } catch { /* Sound is optional. */ }
}
function squawk() { tone("sawtooth", 950, 520, 0.12, 0.05); tone("square", 700, 400, 0.1, 0.02); }
const sounds = {
  chip: () => tone("sine", 880, 1320, 0.12, 0.08),
  gold: () => { tone("triangle", 660, 1320, 0.25, 0.1); tone("sine", 990, 1980, 0.3, 0.06); },
  heart: () => tone("sine", 520, 1040, 0.35, 0.1),
  hit: () => tone("square", 180, 60, 0.25, 0.08),
  streak: () => { tone("triangle", 523, 784, 0.15, 0.08); tone("triangle", 784, 1568, 0.3, 0.06); },
};
function toggleMute() {
  muted = !muted;
  statusEl.textContent = muted ? "Sound off." : "Sound on.";
}

// === AI PROMPT ZONE: PHYSICS & INPUT ===
// Each press is one flap. Ignore a held key's repeat events.
// A resume input leaves the current position intact. A restart creates a new run.
function flap() {
  if (!ctx || document.hidden) return;
  const freshStart = state !== "playing";
  if (state === "paused") {
    state = "playing";
  } else {
    if (state === "over") {
      if (performance.now() - overAt < 350) return;
      reset();
    }
    state = "playing";
    bird.vy = FLAP_STRENGTH;
    squawk();
  }
  if (freshStart) {
    lastTime = performance.now();
    accumulator = 0;
    statusEl.textContent = "Playing. " + lives + " lives. Keep flapping.";
  }
}
function pause() {
  if (state !== "playing") return;
  if (demo) stopDemo(false);
  state = "paused";
  statusEl.textContent = "Paused. Tap, raise your hand or nod up to resume.";
  lastTime = null;
  accumulator = 0;
}
function updateBird(dt) {
  bird.vy = Math.min(MAX_FALL_SPEED, bird.vy + GRAVITY * dt);
  bird.y += bird.vy * dt;
  // The ceiling is forgiving. The floor uses the same life/recovery rule as pipes.
  if (bird.y < bird.r) {
    bird.y = bird.r;
    bird.vy = Math.max(0, bird.vy);
  }
  if (bird.y + bird.r > H - GROUND_H) {
    bird.y = H - GROUND_H - bird.r;
    bird.vy = 0;
    loseLife();
  }
}

// === AI PROMPT ZONE: PIPES / LEVEL ===
// The first two gaps are centered. Later gaps move farther as the run progresses.
// Store each new gap on its pipe. Existing gaps stay still while the player approaches.
// Every visible pipe endcap belongs to its collision box.
function updatePipes(dt) {
  const difficulty = Math.min(1, Math.max(0, (elapsed - RAMP_DELAY) / RAMP_SECONDS));
  const speed = PIPE_SPEED + (MAX_PIPE_SPEED - PIPE_SPEED) * difficulty;
  const last = pipes[pipes.length - 1];
  if (!last || last.x <= W - PIPE_SPACING) {
    const gap = PIPE_GAP + (MIN_PIPE_GAP - PIPE_GAP) * difficulty;
    const drift = 55 + 30 * difficulty;
    if (score > 0) nextCenter = Math.max(gap / 2 + 50,
      Math.min(H - GROUND_H - gap / 2 - 50, nextCenter + (Math.random() * 2 - 1) * drift));
    const roll = Math.random();
    let item = null;
    if (demo) item = nextDemoItem();
    else if (FEATURES.hearts && score >= 3 && lives < MAX_LIVES && roll < HEART_CHANCE) item = "heart";
    else if (FEATURES.gold && roll < HEART_CHANCE + GOLD_CHANCE) item = "gold";
    else if (roll < CHIP_CHANCE) item = "chip";
    pipes.push({ x:last ? W : W + 100, top:nextCenter - gap / 2, gap, passed:false, item });
  }
  for (const pipe of pipes) {
    pipe.x -= speed * dt;
    const overlaps = bird.x + bird.r - 3 > pipe.x && bird.x - bird.r + 3 < pipe.x + PIPE_WIDTH;
    const outsideGap = bird.y - bird.r + 3 < pipe.top || bird.y + bird.r - 3 > pipe.top + pipe.gap;
    if (!recovery && overlaps && outsideGap) {
      loseLife(pipe);
      if (state === "over") return;
    }
    // A chip sits in the middle of the gap, so grabbing it never needs a harder line.
    if (pipe.item) {
      const dx = pipe.x + PIPE_WIDTH / 2 - bird.x, dy = pipe.top + pipe.gap / 2 - bird.y;
      if (dx * dx + dy * dy < (bird.r + 12) ** 2) collect(pipe);
    }
    if (!pipe.passed && pipe.x + PIPE_WIDTH < bird.x - bird.r) {
      pipe.passed = true;
      if (pipe.item === "chip" || pipe.item === "gold") breakStreak();
      addPoint(1);
    }
  }
  pipes = pipes.filter(pipe => pipe.x + PIPE_WIDTH > 0);
  // Night fades in gently once the score reaches NIGHT_SCORE.
  const target = FEATURES.night && score >= NIGHT_SCORE ? 1 : 0;
  night += (target - night) * Math.min(1, dt * 0.8);
  // The sunset deepens as the score climbs toward NIGHT_SCORE.
  const duskTarget = FEATURES.night ? Math.min(1, score / NIGHT_SCORE) : 0;
  dusk += (duskTarget - dusk) * Math.min(1, dt * 0.8);
  if (callout) { callout.t -= dt; if (callout.t <= 0) callout = null; }
  for (const pop of popups) { pop.t -= dt; pop.y -= 40 * dt; }
  popups = popups.filter(pop => pop.t > 0);
}
function collect(pipe) {
  const x = pipe.x + PIPE_WIDTH / 2, y = pipe.top + pipe.gap / 2;
  if (pipe.item === "heart") {
    lives = Math.min(MAX_LIVES, lives + 1);
    popups.push({ x, y, text:"+1 LIFE", t:0.9 });
    statusEl.textContent = "Bonus life. " + lives + " lives.";
    sounds.heart();
  } else {
    const points = pipe.item === "gold" ? GOLD_POINTS : 1;
    chips++;
    addPoint(points);
    if (FEATURES.streak) {
      streak++;
      bestStreak = Math.max(bestStreak, streak);
      if (streak % STREAK_STEP === 0) {
        callout = { text:"STREAK ×" + streak, t:1.3 };
        statusEl.textContent = streak + " chips in a row.";
        sounds.streak();
      }
    }
    popups.push({ x, y, text:"+" + points, t:0.8 });
    sounds[pipe.item]();
  }
  pipe.item = null;
}

// === AI PROMPT ZONE: SCORE & GAME OVER ===
function breakStreak() { streak = 0; }
function addPoint(points) {
  score += points;
  if (score > best && !assisted) {
    best = score;
    try { localStorage.setItem("flappyAiJamBest-seagull-snatch-deluxe", String(best)); } catch { /* Optional. */ }
  }
}
// One contact costs one life. Recovery prevents overlapping obstacles taking all lives.
// Return immediately when the last life is lost. Never score after game over.
function loseLife(pipe) {
  if (recovery || state !== "playing") return;
  lives--;
  breakStreak();
  if (!lives && demo) { stopDemo(true); return; }
  sounds.hit();
  if (!lives) {
    state = "over";
    overAt = performance.now();
    statusEl.textContent = "Flight finished. Score " + score + ", chips " + chips + ". Tap, raise your hand or nod up to try again.";
    return;
  }
  recovery = RECOVERY_SECONDS;
  const next = pipe || pipes.find(item => item.x + PIPE_WIDTH >= bird.x - bird.r);
  bird.y = next ? next.top + next.gap / 2 : 285;
  bird.vy = 0;
  statusEl.textContent = lives + (lives === 1 ? " life left." : " lives left.") + " Keep flapping.";
}

function mix(a, b, t) {
  const pa = parseInt(a.slice(1), 16), pb = parseInt(b.slice(1), 16);
  const ch = shift => Math.round(((pa >> shift) & 255) + (((pb >> shift) & 255) - ((pa >> shift) & 255)) * t);
  return "rgb(" + ch(16) + "," + ch(8) + "," + ch(0) + ")";
}

// === AI PROMPT ZONE: CAMERA FLIGHT ===
// The webcam is read on a small offscreen canvas. No frames leave the device and no library is used.
// Frame differencing finds movement, and the top of the moving area becomes the marker.
// Hand mode: one flap when the marker rises above a fixed trigger line. It re-arms below the re-arm line.
// Head mode: the trigger line follows your resting head height, and one quick upward nod past it is one flap.
// Each flap has a cooldown, so holding still or waving cannot flap again until the marker re-arms.
const CAM_W = 64, CAM_H = 48, CAM_FPS = 30, MAX_MOTION_SHARE = 0.55;
const SENSITIVITY = {
  low:  { diff:40, minPixels:50, handLine:0.34, headRise:0.12 },
  med:  { diff:28, minPixels:30, handLine:0.42, headRise:0.08 },
  high: { diff:18, minPixels:16, handLine:0.50, headRise:0.05 },
};
const MODES = {
  hand: { label:"HAND", cooldown:0.28, smooth:0.6, region:[0, 1], hint:["RAISE HAND ↑", "LOWER HAND ↓"] },
  head: { label:"HEAD", cooldown:0.45, smooth:0.5, region:[0.2, 0.8], hint:["NOD UP ↑", "SETTLE ↓"] },
};
const HUD_X = 10, HUD_Y = 10, HUD_W = 96, HUD_H = 72;
const cam = { on:false, starting:false, video:null, stream:null, bctx:null, prev:null, mode:"hand", sens:"med",
  x:0.5, y:1, base:0.5, seenAt:0, armed:true, lastFlap:0, lastSample:0, message:"", messageUntil:0 };
try {
  const saved = JSON.parse(localStorage.getItem("seagullSnatchCam") || "{}");
  if (MODES[saved.mode]) cam.mode = saved.mode;
  if (SENSITIVITY[saved.sens]) cam.sens = saved.sens;
} catch { /* Settings are optional. */ }
const camTools = document.getElementById("camTools");
const camBtn = document.getElementById("camBtn");
const sensSelect = document.getElementById("sens");
function saveCamSettings() {
  try { localStorage.setItem("seagullSnatchCam", JSON.stringify({ mode:cam.mode, sens:cam.sens })); } catch { /* Optional. */ }
}
function triggerLine() {
  return cam.mode === "hand" ? SENSITIVITY[cam.sens].handLine : Math.max(0.05, cam.base - SENSITIVITY[cam.sens].headRise);
}
function rearmLine() {
  return cam.mode === "hand" ? triggerLine() + 0.16 : cam.base - SENSITIVITY[cam.sens].headRise * 0.35;
}

async function startCam() {
  if (!FEATURES.handCam || cam.on || cam.starting) return;
  if (!window.isSecureContext || !navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
    camNotice("Camera not available here. Tap or click to fly.", 6);
    return;
  }
  cam.starting = true;
  updateCamControls();
  try {
    const stream = await navigator.mediaDevices.getUserMedia({ video:true });
    const video = document.createElement("video");
    video.muted = true; video.playsInline = true; video.srcObject = stream;
    await video.play();
    const buffer = document.createElement("canvas");
    buffer.width = CAM_W; buffer.height = CAM_H;
    Object.assign(cam, { on:true, video, stream, bctx:buffer.getContext("2d", { willReadFrequently:true }), prev:null, armed:true, y:0.6, base:0.6 });
    const track = stream.getVideoTracks()[0];
    if (track) track.addEventListener("ended", () => stopCam("Camera disconnected. Tap or click to fly."));
    camNotice(cam.mode === "head" ? "Camera on. Nod up quickly to flap." : "Camera on. Raise your hand above the line to flap.", 4);
  } catch (err) {
    const name = err && err.name;
    camNotice(name === "NotAllowedError" || name === "SecurityError" ? "Camera blocked. Tap or click to fly instead."
      : name === "NotFoundError" || name === "OverconstrainedError" ? "No camera found. Tap or click to fly instead."
      : "Camera could not start. Tap or click to fly instead.", 6);
  } finally {
    cam.starting = false;
    updateCamControls();
  }
}
function stopCam(message) {
  if (cam.stream) cam.stream.getTracks().forEach(track => track.stop());
  Object.assign(cam, { on:false, stream:null, video:null, prev:null });
  updateCamControls();
  camNotice(message || "Camera off. Tap or click to fly.", 3);
}
function toggleCam() { if (cam.on) stopCam(); else startCam(); }
function sampleCam(now) {
  if (demo || !cam.on || !cam.video || cam.video.readyState < 2 || now - cam.lastSample < 1000 / CAM_FPS) return;
  cam.lastSample = now;
  const c = cam.bctx, sens = SENSITIVITY[cam.sens], mode = MODES[cam.mode];
  c.save(); c.translate(CAM_W, 0); c.scale(-1, 1); c.drawImage(cam.video, 0, 0, CAM_W, CAM_H); c.restore();
  let data;
  try { data = c.getImageData(0, 0, CAM_W, CAM_H).data; } catch { return; }
  const gray = new Uint8Array(CAM_W * CAM_H);
  for (let i = 0, j = 0; i < gray.length; i++, j += 4) gray[i] = (data[j] * 77 + data[j + 1] * 150 + data[j + 2] * 29) >> 8;
  const prev = cam.prev;
  cam.prev = gray;
  if (!prev) return;
  const x0 = Math.floor(mode.region[0] * CAM_W), x1 = Math.ceil(mode.region[1] * CAM_W);
  const rows = new Uint16Array(CAM_H), rowX = new Float32Array(CAM_H);
  let moving = 0;
  for (let y = 0; y < CAM_H; y++) {
    for (let x = x0; x < x1; x++) {
      const i = y * CAM_W + x;
      if (Math.abs(gray[i] - prev[i]) > sens.diff) { rows[y]++; rowX[y] += x; moving++; }
    }
  }
  if (moving < sens.minPixels || moving > (x1 - x0) * CAM_H * MAX_MOTION_SHARE) return;
  // The top fifth of the moving pixels is where a raised hand or the top of the head is.
  let count = 0, sumX = 0, row = 0;
  for (; row < CAM_H; row++) {
    count += rows[row]; sumX += rowX[row];
    if (count >= moving * 0.2) break;
  }
  cam.x += (sumX / count / CAM_W - cam.x) * mode.smooth;
  cam.y += ((row + 0.5) / CAM_H - cam.y) * mode.smooth;
  cam.seenAt = now;
  // In Head mode the resting height drifts slowly toward where the head is, so the line follows you.
  if (cam.mode === "head" && cam.armed) cam.base += (cam.y - cam.base) * 0.08;
  if (cam.armed && cam.y < triggerLine() && now - cam.lastFlap > mode.cooldown * 1000) {
    cam.armed = false;
    cam.lastFlap = now;
    flap();
  } else if (!cam.armed && cam.y > rearmLine()) {
    cam.armed = true;
  }
}

}

  demo = null;
  updateDemoButton();
  if (finished) {
    reset();
    state = "start";
    statusEl.textContent = "Demo finished. Tap to play.";
  } else {
    statusEl.textContent = "You have control. " + lives + " lives.";
  }
}
function nextDemoItem() {
  let item = DEMO_ITEMS[demo.items++ % DEMO_ITEMS.length];
  if (item === "gold" && !FEATURES.gold) item = "chip";
  if (item === "heart" && (!FEATURES.hearts || lives >= MAX_LIVES)) item = "chip";
  return item;
}
// Runs inside the fixed step, so the demo plays the same at any refresh rate.
function demoStep(dt) {
  demo.t -= dt;
  demo.sinceFlap += dt;
  if (demo.t <= 0) { stopDemo(true); return; }
  const next = pipes.find(pipe => pipe.x + PIPE_WIDTH >= bird.x - bird.r);
  const target = next ? next.top + next.gap / 2 : H * 0.45;
  // Flap once the gull is falling and has dropped a little below the gap centre.
  // Each flap lifts it about 46px, so it bobs gently around the centre of the gap.
  if (bird.vy > 0 && bird.y > target + 20 && demo.sinceFlap > 0.18) {
    bird.vy = FLAP_STRENGTH;
    squawk();
    demo.sinceFlap = 0;
    demo.pulseAt = elapsed;
  }
}
function drawDemoHud(light) {
  const mode = demo.t > DEMO_SECONDS / 2 ? "hand" : "head";
  const info = MODES[mode], sens = SENSITIVITY[cam.sens];
  const trigger = mode === "hand" ? sens.handLine : 0.55 - sens.headRise;
  const rearm = mode === "hand" ? sens.handLine + 0.16 : 0.55 - sens.headRise * 0.35;
  const rest = mode === "hand" ? Math.min(0.92, rearm + 0.12) : 0.6;
  const since = elapsed - demo.pulseAt;
  // The marker pops just above the line as the flap fires, then settles back below the re-arm line.
  let y = rest;
  if (since < 0.08) y = trigger + 0.04 - (since / 0.08) * 0.12;
  else if (since < 0.4) y = trigger - 0.08 + ((since - 0.08) / 0.32) * (rest - trigger + 0.08);
  const cooling = since < info.cooldown;
  const x = HUD_X, top = HUD_Y;
  ctx.fillStyle = COLORS.navy;
  ctx.fillRect(x - 2, top - 2, HUD_W + 4, HUD_H + 4);
  ctx.fillStyle = "#0B2B40";
  ctx.fillRect(x, top, HUD_W, HUD_H);
  // A simple silhouette stands in for the webcam picture.
  ctx.fillStyle = "rgba(247,245,240,0.28)";
  if (mode === "hand") {
    ctx.beginPath(); ctx.arc(x + 42, top + 50, 11, 0, Math.PI * 2); ctx.fill();
    ctx.fillRect(x + 26, top + 62, 32, 12);
    ctx.fillRect(x + 63, top + y * HUD_H, 8, HUD_H - y * HUD_H);
    ctx.beginPath(); ctx.arc(x + 67, top + y * HUD_H, 7, 0, Math.PI * 2); ctx.fill();
  } else {
    ctx.beginPath(); ctx.arc(x + HUD_W / 2, top + y * HUD_H + 10, 13, 0, Math.PI * 2); ctx.fill();
    ctx.fillRect(x + HUD_W / 2 - 20, top + y * HUD_H + 24, 40, 30);
  }
  const lineY = top + HUD_H * trigger;
  ctx.strokeStyle = COLORS.beak; ctx.lineWidth = 2;
  ctx.setLineDash([5, 4]);
  ctx.beginPath(); ctx.moveTo(x, lineY); ctx.lineTo(x + HUD_W, lineY); ctx.stroke();
  ctx.setLineDash([]);
  ctx.strokeStyle = "rgba(247,245,240,0.45)"; ctx.lineWidth = 1;
  ctx.beginPath(); ctx.moveTo(x, top + HUD_H * rearm); ctx.lineTo(x + HUD_W, top + HUD_H * rearm); ctx.stroke();
  const flash = since < 0.2;
  ctx.fillStyle = flash ? COLORS.beak : cooling ? COLORS.stripe : "#3DDC84";
  const markerX = mode === "hand" ? 67 : HUD_W / 2;
  ctx.beginPath(); ctx.arc(x + markerX, top + y * HUD_H, flash ? 7 : 5, 0, Math.PI * 2); ctx.fill();
  ctx.textAlign = "left";
  ctx.font = "bold 10px system-ui, sans-serif";
  ctx.fillStyle = "rgba(7,59,92,0.85)";
  ctx.fillRect(x, top, 34, 14); ctx.fillRect(x + HUD_W - 34, top, 34, 14);
  ctx.fillStyle = COLORS.cream;
  ctx.fillText(info.label, x + 4, top + 11);
  ctx.fillStyle = COLORS.beak;
  ctx.fillText("DEMO", x + HUD_W - 31, top + 11);
  // Cooldown bar: the next flap can only fire once it is full.
  ctx.fillStyle = "rgba(7,59,92,0.35)";
  ctx.fillRect(x, top + HUD_H + 4, HUD_W, 4);
  ctx.fillStyle = cooling ? COLORS.stripe : "#3DDC84";
  ctx.fillRect(x, top + HUD_H + 4, HUD_W * Math.min(1, since / info.cooldown), 4);
  ctx.fillStyle = light ? COLORS.cream : COLORS.navy;
  ctx.fillText(cooling ? "COOLDOWN" : info.hint[0], x, top + HUD_H + 20);
}

// === AI PROMPT ZONE: GAME LOOP ===
// Fixed steps make the same inputs play the same at 30, 60 and 144Hz.
// Clamp long frames and pause on focus loss so a tab switch cannot cost a life.
function loop(timestamp) {
  if (lastTime !== null && state === "playing") {
    accumulator += Math.min(0.05, Math.max(0, (timestamp - lastTime) / 1000));
    while (accumulator + 1e-9 >= STEP && state === "playing") {
      recovery = Math.max(0, recovery - STEP);
      elapsed += STEP;
      if (demo) { demoStep(STEP); if (state !== "playing") break; }
      updateBird(STEP);
      if (state === "playing") updatePipes(STEP);
      accumulator = Math.max(0, accumulator - STEP);
    }
  }
  lastTime = timestamp;
  if (FEATURES.handCam) sampleCam(timestamp);
  drawBackground(); drawPipes(); drawBird(); drawUI();
  if (demo) drawDemoBanner();
  requestAnimationFrame(loop);
}
addEventListener("keydown", event => {
  if (event.defaultPrevented || event.altKey || event.ctrlKey || event.metaKey) return;
  if (event.target.closest?.("a, button, input, textarea, select, [contenteditable]")) return;
  // Space and Arrow Up do not flap in this build. Camera movement, mouse and touch do.
  const isMute = FEATURES.sound && event.code === "KeyM";
  const isCam = FEATURES.handCam && event.code === "KeyC";
  if (event.code !== "KeyP" && !isMute && !isCam) return;
  event.preventDefault();
  if (event.repeat) return;
  if (isMute) { toggleMute(); return; }
  if (isCam) { toggleCam(); return; }
  if (demo) { pause(); return; }
  if (state === "paused") flap();
  else pause();
});
