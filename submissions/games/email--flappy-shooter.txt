// === AI PROMPT ZONE: PHYSICS & INPUT ===
// Each press is one flap. Ignore a held key's repeat events.
// A resume input leaves the current position intact. A restart creates a new run.
// The bird fires automatically to the right on a steady timer.
function flap() {
  if (!ctx || document.hidden) return;
  const freshStart = state !== "playing";
  if (state === "paused") {
    state = "playing";
  } else {
    if (state === "over") {
      if (performance.now() - overAt < 350) return;
      reset();
    }
    state = "playing";
    bird.vy = FLAP_STRENGTH;
  }
  if (freshStart) {
    lastTime = performance.now();
    accumulator = 0;
    statusEl.textContent = "Playing. " + lives + " lives. Flap to dodge, you fire automatically.";
  }
}
function pause() {

  state = "paused";
  statusEl.textContent = "Paused. Tap or press Space to resume.";
  lastTime = null;
  accumulator = 0;
}
function updateBird(dt) {
  bird.vy = Math.min(MAX_FALL_SPEED, bird.vy + GRAVITY * dt);
  bird.y += bird.vy * dt;
  fireTimer -= dt;
  if (fireTimer <= 0) {
    const w = WEAPONS[weapon], pattern = power.spread ? w.triple : w;
    fireTimer += w.every * (power.rapid ? RAPID_FIRE_MULT : 1);
    volley++;
    for (let i = 0; i < pattern.count; i++) {
      const even = pattern.count > 1 ? (i / (pattern.count - 1)) * 2 - 1 : 0;
      const a = even * pattern.spread + (Math.random() * 2 - 1) * pattern.jitter;
      shots.push({ volley, x:bird.x + bird.r + 6, y:bird.y + 2, vx:Math.cos(a) * w.speed, vy:Math.sin(a) * w.speed,
        r:w.r, dmg:w.dmg, bossDmg:w.bossDmg, bossCooldown:w.every * 0.8, life:w.range, homing:w.homing || 0, kind:weapon });
    }
  }
  // The ceiling is forgiving. The floor uses the same life/recovery rule as enemy hits,
  // except a shielded bird bounces off it.
  if (bird.y < bird.r) {
    bird.y = bird.r;
    bird.vy = Math.max(0, bird.vy);
  }
  if (bird.y + bird.r > H - GROUND_H) {
    bird.y = H - GROUND_H - bird.r;
    if (power.shield) {
      bird.vy = FLAP_STRENGTH * 0.8;
    } else {
      bird.vy = 0;
      loseLife(true);
    }
  }
}

// === AI PROMPT ZONE: PIPES / LEVEL ===
// Pipes are replaced by enemy drones that fly in from the right and shoot at the bird.
// The first SAFE_ENEMIES drones hold fire. Later drones arrive faster, fire more often and can be armoured.
// Armoured (purple) drones take TOUGH_HP hits and fire a 3-shot fan.
// Enemy shots aim at the bird but are capped to a shallow angle, and drones only fire with
// room to react, so every shot can be dodged by flapping.
function lerp(a, b, t) { return a + (b - a) * t; }
function hits(a, b, dist) { const dx = a.x - b.x, dy = a.y - b.y; return dx * dx + dy * dy < dist * dist; }
function spawnEnemy(difficulty) {
  const tough = spawned >= 6 && Math.random() < 0.1 + 0.3 * difficulty;
  const r = tough ? 20 : 15;
  const top = 40 + r, bottom = H - GROUND_H - 40 - r;
  const safe = spawned < SAFE_ENEMIES;
  const baseY = safe ? 285 - spawned * 60 : top + Math.random() * (bottom - top);
  enemies.push({ x:W + r + 10, y:baseY, baseY, r, tough, hp:tough ? TOUGH_HP : 1, points:tough ? TOUGH_POINTS : 1,
    speed:lerp(ENEMY_SPEED, MAX_ENEMY_SPEED, difficulty) * (0.9 + Math.random() * 0.2),
    amp:safe ? 8 : 10 + Math.random() * 25, phase:Math.random() * Math.PI * 2, age:0, flash:0,
    fireTimer:safe ? Infinity : 0.6 + Math.random() * 0.8 });
  spawned++;
}
function spawnBoss() {
  const hp = BOSS_HP + BOSS_HP_GROWTH * bossCount++;
  enemies.push({ x:W + BOSS_R + 10, y:275, baseY:275, r:BOSS_R, boss:true, tough:true, hp, maxHp:hp,
    points:BOSS_POINTS, speed:60, amp:140, phase:0, age:0, flash:0, fireTimer:1.5, lastVolley:-1, hitCooldown:0,
    attack:"fan", switchChance:BOSS_SWITCH_CHANCE });
  nextBossAt = Infinity;
  banner = { text:"BOSS INCOMING!", color:COLORS.eye, t:1.5 };
  statusEl.textContent = "Boss drone incoming.";
}
// Homing rockets turn toward the nearest drone ahead of them, limited by their turn rate.
function steerRocket(s, dt) {
  let target = null, bestD = Infinity;
  for (const e of enemies) {
    if (e.hp <= 0 || e.x < s.x - 10) continue;
    const d = (e.x - s.x) ** 2 + (e.y - s.y) ** 2;
    if (d < bestD) { bestD = d; target = e; }
  }
  if (!target) return;
  const speed = Math.hypot(s.vx, s.vy), current = Math.atan2(s.vy, s.vx);
  let diff = Math.atan2(target.y - s.y, target.x - s.x) - current;
  diff = Math.atan2(Math.sin(diff), Math.cos(diff));
  const a = current + Math.max(-s.homing * dt, Math.min(s.homing * dt, diff));
  s.vx = Math.cos(a) * speed;
  s.vy = Math.sin(a) * speed;
}
function fireWall(e) {
  const floor = H - GROUND_H;
  const center = Math.max(BOSS_WALL_GAP / 2 + 20, Math.min(floor - BOSS_WALL_GAP / 2 - 20,
    bird.y + (Math.random() * 2 - 1) * BOSS_WALL_REACH));
  for (let y = ENEMY_SHOT_R; y < floor; y += BOSS_WALL_SPACING) {
    if (Math.abs(y - center) < BOSS_WALL_GAP / 2) continue;
    enemyShots.push({ x:e.x - e.r, y, vx:-BOSS_WALL_SPEED, vy:0 });
  }
  e.fireTimer *= BOSS_WALL_PAUSE;
}
function nextBossAttack(e) {
  if (Math.random() < e.switchChance) {
    e.attack = e.attack === "fan" ? "wall" : "fan";
    e.switchChance = BOSS_SWITCH_CHANCE;
  } else {
    e.switchChance = Math.min(1, e.switchChance + BOSS_SWITCH_RAMP);
  }
}
function updateEnemies(dt) {
  const difficulty = Math.min(1, Math.max(0, (elapsed - RAMP_DELAY) / RAMP_SECONDS));
  const bossAlive = enemies.some(e => e.boss);
  if (!bossAlive && score >= nextBossAt) spawnBoss();
  if (!bossAlive) spawnTimer -= dt;
  if (spawnTimer <= 0) {
    spawnEnemy(difficulty);
    spawnTimer = lerp(SPAWN_EVERY, MIN_SPAWN_EVERY, difficulty) * (0.85 + Math.random() * 0.3);
  }
  const shotSpeed = lerp(ENEMY_SHOT_SPEED, MAX_ENEMY_SHOT_SPEED, difficulty);
  for (const e of enemies) {
    e.age += dt;
    e.flash = Math.max(0, e.flash - dt);
    if (e.boss) {
      e.hitCooldown = Math.max(0, e.hitCooldown - dt);
      // The boss flies in, then holds near the right edge and sweeps up and down.
      e.x = Math.max(W - 70, e.x - e.speed * dt);
      e.y = e.baseY + Math.sin(e.age * 1.1) * e.amp;
    } else {
      e.x -= e.speed * dt;
      e.y = e.baseY + Math.sin(e.age * 2.2 + e.phase) * e.amp;
    }
    e.fireTimer -= dt;
    if (e.fireTimer <= 0) {
      e.fireTimer = e.boss ? lerp(BOSS_FIRE_EVERY, MIN_BOSS_FIRE_EVERY, difficulty)
        : lerp(ENEMY_FIRE_EVERY, MIN_ENEMY_FIRE_EVERY, difficulty)
          * (e.tough ? TOUGH_FIRE_SLOWER : 1) * (0.8 + Math.random() * 0.4);
      if (e.boss && e.attack === "wall" && e.x < W - e.r) {
        fireWall(e);
      } else if (e.x < W - e.r && e.x - bird.x > 110) {
        const aim = Math.atan2(bird.y - e.y, e.x - bird.x);
        const angle = Math.max(-MAX_AIM_ANGLE, Math.min(MAX_AIM_ANGLE, aim));
        const fan = e.boss ? Array.from({ length:BOSS_FAN }, (_, i) => (i - (BOSS_FAN - 1) / 2) * BOSS_SPREAD)
          : e.tough ? [-TOUGH_SPREAD, 0, TOUGH_SPREAD] : [0];
        for (const f of fan) {
          enemyShots.push({ x:e.x - e.r, y:e.y,
            vx:-Math.cos(angle + f) * shotSpeed, vy:Math.sin(angle + f) * shotSpeed });
        }
      }
      if (e.boss) nextBossAttack(e);
    }
    if (e.hp > 0 && hits(e, bird, e.r + bird.r - 4)) {
      if (e.boss) {
        // Ramming the boss never kills it. Without a shield it costs a life.
        if (!power.shield && !recovery) {
          loseLife(false);
          if (state === "over") return;
        }
      } else if (power.shield) {
        e.hp = 0;
        killEnemy(e);
      } else if (!recovery) {
        e.hp = 0;
        burst(e.x, e.y, e.tough ? COLORS.enemyTough : COLORS.enemy, 10);
        loseLife(false);
        if (state === "over") return;
      }
    }
  }
  for (const s of shots) {
    s.life -= dt;
    if (s.homing) steerRocket(s, dt);
    s.x += s.vx * dt;
    s.y += s.vy * dt;
    for (const e of enemies) {
      if (s.done || e.hp <= 0 || !hits(s, e, e.r + s.r)) continue;
      s.done = true;
      if (e.boss) {
        // Extra shots from the same volley, or inside the cooldown, are absorbed with no damage.
        if (s.volley === e.lastVolley || e.hitCooldown > 0) continue;
        e.lastVolley = s.volley;
        e.hitCooldown = Math.max(BOSS_HIT_COOLDOWN, s.bossCooldown);
      }
      e.hp -= e.boss ? s.bossDmg : s.dmg;
      e.flash = 0.08;
      if (e.hp <= 0) killEnemy(e);
    }
  }
  for (const b of enemyShots) {
    b.x += b.vx * dt;
    b.y += b.vy * dt;
    if (!hits(b, bird, bird.r - 3 + ENEMY_SHOT_R)) continue;
    if (power.shield) {
      b.done = true;
      burst(b.x, b.y, POWERUPS.shield.color, 4);
    } else if (!recovery) {
      b.done = true;
      loseLife(false);
      if (state === "over") return;
    }
  }
  shots = shots.filter(s => !s.done && s.life > 0 && s.x > -10 && s.x < W + 10 && s.y > -10 && s.y < H - GROUND_H);
  enemies = enemies.filter(e => e.hp > 0 && e.x > -e.r - 10);
  enemyShots = enemyShots.filter(b => !b.done && b.x > -10 && b.y > -10 && b.y < H - GROUND_H);
}
function killEnemy(e) {
  addScore(e.points);
  burst(e.x, e.y, e.tough ? COLORS.enemyTough : COLORS.enemy, e.boss ? 50 : e.tough ? 18 : 10);
  maybeDropPowerup(e);
  if (e.boss) {
    nextBossAt = score + BOSS_EVERY;
    spawnTimer = 1.5;
    banner = { text:"BOSS DOWN! +" + e.points, color:COLORS.navy, t:1.5 };
    statusEl.textContent = "Boss destroyed. Score " + score + ".";
  }
}

// === AI PROMPT ZONE: POWER-UPS ===
// Destroyed drones sometimes drop a power-up (armoured drones more often). Fly into it to collect it.
// The drop chance starts at DROP_CHANCE (TOUGH_DROP_CHANCE for armoured drones) and rises by DROP_RAMP
// for every kill without a drop, then resets when something drops. Bosses always drop.
// Timed power-ups refresh to full when collected again. Add a new one to POWERUPS in CONFIG,
// then handle it in collectPowerup and wherever it changes play (firing, hits, drawing).
function maybeDropPowerup(e) {
  if (state !== "playing") return;
  const base = e.boss ? 1 : e.tough ? TOUGH_DROP_CHANCE : DROP_CHANCE;
  if (Math.random() >= base + DROP_RAMP * dryKills) {
    dryKills++;
    return;
  }
  dryKills = 0;
  const options = Object.keys(POWERUPS).filter(k => (k !== "life" || lives < START_LIVES) && k !== weapon);
  let roll = Math.random() * options.reduce((total, k) => total + POWERUPS[k].weight, 0);
  const type = options.find(k => (roll -= POWERUPS[k].weight) < 0) || options[0];
  const y = Math.max(30, Math.min(H - GROUND_H - 30, e.y));
  pickups.push({ type, x:e.x, y, baseY:y, age:0 });
}
function collectPowerup(p) {
  const def = POWERUPS[p.type];
  if (p.type === "life") lives = Math.min(START_LIVES, lives + 1);
  else if (def.weapon) weapon = p.type;
  else power[p.type] = def.seconds;
  banner = { text:def.label + "!", color:def.color, t:1.2 };
  burst(p.x, p.y, def.color, 10);
  statusEl.textContent = def.label + ". " + lives + (lives === 1 ? " life." : " lives.");
}
function updatePowerups(dt) {
  for (const k in power) power[k] = Math.max(0, power[k] - dt);
  if (banner && (banner.t -= dt) <= 0) banner = null;
  for (const p of pickups) {
    p.age += dt;
    p.x -= PICKUP_SPEED * dt;
    p.y = p.baseY + Math.sin(p.age * 4) * 6;
    if (hits(p, bird, PICKUP_R + bird.r)) {
      p.done = true;
      collectPowerup(p);
    }
  }
  pickups = pickups.filter(p => !p.done && p.x > -PICKUP_R);
}
function burst(x, y, color, count) {
  for (let i = 0; i < count; i++) {
    const a = Math.random() * Math.PI * 2, sp = 60 + Math.random() * 140;
    particles.push({ x, y, vx:Math.cos(a) * sp, vy:Math.sin(a) * sp, life:0.45, max:0.45, color });
  }
}
function updateParticles(dt) {
  for (const p of particles) {
    p.life -= dt;
    p.vy += 300 * dt;
    p.x += p.vx * dt;
    p.y += p.vy * dt;
  }
  particles = particles.filter(p => p.life > 0);
}

// === AI PROMPT ZONE: SCORE & GAME OVER ===
// Each drone destroyed earns its points (1, armoured drones TOUGH_POINTS). One hit costs one life.
// Recovery clears incoming fire and prevents one volley taking all lives.
// Return immediately when the last life is lost. Never score after game over.
function addScore(points) {
  if (state !== "playing") return;
  score += points;
  if (score > best) {
    best = score;
    try { localStorage.setItem(BEST_KEY, String(best)); } catch { /* Optional. */ }
  }
}
function loseLife(fromFloor) {
  if (recovery || state !== "playing") return;
  lives--;
  burst(bird.x, bird.y, COLORS.bird, 12);
  if (!lives) {
    state = "over";
    overAt = performance.now();
    statusEl.textContent = "Flight finished. Score " + score + ". Tap or press Space to try again.";
    return;
  }
  recovery = RECOVERY_SECONDS;
  enemyShots = [];
  if (fromFloor) bird.y = 285;
  bird.vy = 0;
  statusEl.textContent = lives + (lives === 1 ? " life left." : " lives left.") + " Keep flapping.";
}

// === AI PROMPT ZONE: GRAPHICS ===
// Draw with shapes. Keep controls and the score legible over any new artwork.
function drawBackground() {
  ctx.fillStyle = COLORS.cream;
  ctx.fillRect(0, 0, W, H);
  ctx.fillStyle = "rgba(7,59,92,0.08)";
  for (let i = 0; i < 4; i++) {
    const x = ((i * 140 - elapsed * 40) % (W + 120) + W + 120) % (W + 120) - 60;
    ctx.beginPath();
    ctx.arc(x, 80 + i * 45, 30, 0, Math.PI * 2);
    ctx.arc(x + 30, 75 + i * 45, 22, 0, Math.PI * 2);
    ctx.fill();
  }
}
function drawEnemies() {
  for (const e of enemies) {
    ctx.save();
    ctx.translate(e.x, e.y);
    const wing = Math.sin(e.age * 18) * 5;
    ctx.fillStyle = e.flash ? "#fff" : e.tough ? COLORS.enemyTough : COLORS.enemy;
    ctx.beginPath(); ctx.moveTo(-4, -e.r + 4); ctx.lineTo(8, -e.r - 8 - wing); ctx.lineTo(12, -e.r + 6); ctx.fill();
    ctx.beginPath(); ctx.moveTo(-4, e.r - 4); ctx.lineTo(8, e.r + 8 + wing); ctx.lineTo(12, e.r - 6); ctx.fill();
    ctx.beginPath(); ctx.arc(0, 0, e.r, 0, Math.PI * 2); ctx.fill();
    ctx.fillStyle = COLORS.cream;
    ctx.beginPath(); ctx.arc(-e.r * 0.35, -2, e.r * 0.4, 0, Math.PI * 2); ctx.fill();
    ctx.fillStyle = COLORS.eye;
    ctx.beginPath(); ctx.arc(-e.r * 0.45, -2, e.r * 0.22, 0, Math.PI * 2); ctx.fill();
    if (e.tough && !e.boss) {
      ctx.fillStyle = COLORS.cream;
      for (let i = 0; i < e.hp; i++) ctx.fillRect(-9 + i * 7, e.r * 0.45, 5, 4);
    }
    ctx.restore();
  }
}
function drawShots() {
  for (const s of shots) {
    ctx.save();
    ctx.translate(s.x, s.y);
    ctx.rotate(Math.atan2(s.vy, s.vx));
    if (s.kind === "rocket") {
      ctx.fillStyle = COLORS.navy; ctx.fillRect(-12, -3, 12, 6);
      ctx.fillStyle = COLORS.playerShot;
      ctx.beginPath(); ctx.moveTo(0, -3); ctx.lineTo(6, 0); ctx.lineTo(0, 3); ctx.fill();
      ctx.fillStyle = COLORS.bird; ctx.fillRect(-17, -2, 5, 4);
    } else {
      ctx.fillStyle = COLORS.playerShot;
      ctx.fillRect(-s.r * 2.5, -1.5, s.r * 2.5, 3);
      ctx.beginPath(); ctx.arc(0, 0, s.r, 0, Math.PI * 2); ctx.fill();
    }
    ctx.restore();
  }
  ctx.strokeStyle = COLORS.navy;
  ctx.lineWidth = 1.5;
  for (const b of enemyShots) {
    ctx.fillStyle = COLORS.enemyShot;
    ctx.beginPath(); ctx.arc(b.x, b.y, ENEMY_SHOT_R, 0, Math.PI * 2); ctx.fill(); ctx.stroke();
  }
}
function drawIcon(def, x, y, r) {
  // Weapons are square icons, timed power-ups are round.
  ctx.fillStyle = def.color;
  ctx.beginPath();
  if (def.weapon) ctx.rect(x - r, y - r, r * 2, r * 2);
  else ctx.arc(x, y, r, 0, Math.PI * 2);
  ctx.fill();
  ctx.strokeStyle = COLORS.cream; ctx.lineWidth = 2; ctx.stroke();
  ctx.fillStyle = COLORS.cream;
  ctx.font = "bold " + Math.round(r * (def.icon.length > 1 ? 0.95 : 1.3)) + "px system-ui, sans-serif";
  ctx.textAlign = "center"; ctx.textBaseline = "middle";
  ctx.fillText(def.icon, x, y + 1);
  ctx.textBaseline = "alphabetic";
}
function drawPickups() {
  for (const p of pickups) {
    const def = POWERUPS[p.type];
    ctx.strokeStyle = def.color;
    ctx.globalAlpha = 0.35 + 0.25 * Math.sin(p.age * 8);
    ctx.lineWidth = 3;
    ctx.beginPath(); ctx.arc(p.x, p.y, PICKUP_R + 5, 0, Math.PI * 2); ctx.stroke();
    ctx.globalAlpha = 1;
    drawIcon(def, p.x, p.y, PICKUP_R);
  }
}
function drawParticles() {
  for (const p of particles) {
    ctx.globalAlpha = Math.max(0, p.life / p.max);
    ctx.fillStyle = p.color;
    ctx.fillRect(p.x - 2, p.y - 2, 4, 4);
  }
  ctx.globalAlpha = 1;
}
function drawBird() {
  ctx.save();
  ctx.translate(bird.x, bird.y);
  // The shield blinks during its last 1.5 seconds.
  if (power.shield && (power.shield > 1.5 || Math.floor(power.shield * 8) % 2)) {
    ctx.fillStyle = "rgba(47,128,237,0.15)";
    ctx.strokeStyle = POWERUPS.shield.color;
    ctx.lineWidth = 3;
    ctx.beginPath(); ctx.arc(0, 0, bird.r + 10, 0, Math.PI * 2); ctx.fill(); ctx.stroke();
  }
  ctx.rotate(Math.max(-0.4, Math.min(0.8, bird.vy / 500)));
  if (recovery) {
    ctx.strokeStyle = COLORS.navy;
    ctx.lineWidth = 2;
    ctx.beginPath(); ctx.arc(0, 0, bird.r + 7, 0, Math.PI * 2); ctx.stroke();
  }
  ctx.fillStyle = COLORS.bird;
  ctx.beginPath(); ctx.arc(0, 0, bird.r, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = "#fff";
  ctx.beginPath(); ctx.arc(6, -5, 6, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = COLORS.navy;
  ctx.beginPath(); ctx.arc(8, -5, 3, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = COLORS.beak;
  ctx.beginPath(); ctx.moveTo(12, 0); ctx.lineTo(24, 4); ctx.lineTo(12, 8); ctx.fill();
  ctx.restore();
}
function drawText(text, y, size, color) {
  ctx.fillStyle = color || COLORS.navy;
  ctx.font = "bold " + size + "px system-ui, sans-serif";
  ctx.textAlign = "center";
  ctx.fillText(String(text), W / 2, y);
}
function drawUI() {
  drawText(score, 58, 38);
  if (weapon !== "blaster") {
    drawIcon(POWERUPS[weapon], W - 22, 22, 9);
    ctx.fillStyle = COLORS.navy;
    ctx.font = "bold 11px system-ui, sans-serif";
    ctx.textAlign = "right";
    ctx.fillText(WEAPONS[weapon].label, W - 38, 26);
  }
  let row = 0;
  for (const k of ["rapid", "spread", "shield"]) {
    if (!power[k]) continue;
    const def = POWERUPS[k], y = 22 + row++ * 24;
    drawIcon(def, 22, y, 9);
    ctx.fillStyle = "rgba(7,59,92,0.15)";
    ctx.fillRect(36, y - 3, 56, 6);
    ctx.fillStyle = def.color;
    ctx.fillRect(36, y - 3, 56 * power[k] / def.seconds, 6);
  }
  const boss = enemies.find(e => e.boss);
  if (boss) {
    ctx.fillStyle = "rgba(7,59,92,0.15)";
    ctx.fillRect(110, 72, 180, 8);
    ctx.fillStyle = COLORS.eye;
    ctx.fillRect(110, 72, 180 * Math.max(0, boss.hp) / boss.maxHp, 8);
  }
  if (banner) {
    ctx.globalAlpha = Math.min(1, banner.t * 2);
    drawText(banner.text, 100, 20, banner.color);
    ctx.globalAlpha = 1;
  }
  ctx.fillStyle = COLORS.ground;
  ctx.fillRect(0, H - GROUND_H, W, GROUND_H);
  ctx.fillStyle = "rgba(7,59,92,0.18)";
  for (let i = 0; i < 11; i++) {
    const x = ((i * 40 - elapsed * 90) % 440 + 440) % 440 - 40;
    ctx.fillRect(x, H - GROUND_H, 20, 4);
  }
  drawText("Lives " + lives + " / " + START_LIVES + "   ·   Best " + best, H - 18, 16);
  if (state === "playing") return;
  ctx.fillStyle = COLORS.cream;
  ctx.fillRect(24, 175, W - 48, 205);
  ctx.strokeStyle = COLORS.navy; ctx.lineWidth = 2;
  ctx.strokeRect(24, 175, W - 48, 205);
  if (state === "start") {
    drawText("FLAPPY AI JAM", 225, 32);
    drawText("Flap to dodge. You fire automatically.", 268, 16);
    drawText("Shoot the drones before they shoot you.", 295, 16);
    drawText("Tap or press Space to start", 335, 18);
    drawText("Grab power-ups. It gets harder as you go.", 362, 14);
  } else if (state === "paused") {
    drawText("PAUSED", 235, 32);
    drawText("Tap or press Space to resume", 305, 18);
  } else {
    drawText("FLIGHT OVER", 225, 32);
    drawText("Score " + score + "   ·   Best " + best, 275, 22);
    drawText("Tap or press Space to try again", 330, 18);
  }
}

// === AI PROMPT ZONE: GAME LOOP ===
// Fixed steps make the same inputs play the same at 30, 60 and 144Hz.
// Clamp long frames and pause on focus loss so a tab switch cannot cost a life.
function loop(timestamp) {
  if (lastTime !== null && state === "playing") {
    accumulator += Math.min(0.05, Math.max(0, (timestamp - lastTime) / 1000));
    while (accumulator + 1e-9 >= STEP && state === "playing") {
      recovery = Math.max(0, recovery - STEP);
      elapsed += STEP;
      updateBird(STEP);
      if (state === "playing") updateEnemies(STEP);
      if (state === "playing") updatePowerups(STEP);
      updateParticles(STEP);
      accumulator = Math.max(0, accumulator - STEP);
    }
  }
  lastTime = timestamp;
  drawBackground(); drawPickups(); drawEnemies(); drawShots(); drawBird(); drawParticles(); drawUI();
  requestAnimationFrame(loop);
}
addEventListener("keydown", event => {
  if (event.defaultPrevented || event.altKey || event.ctrlKey || event.metaKey) return;
  if (event.target.closest?.("a, button, input, textarea, select, [contenteditable]")) return;
  const isFlap = event.code === "Space" || event.code === "ArrowUp";
  if (!isFlap && event.code !== "KeyP") return;
  event.preventDefault();
  if (event.repeat) return;
  if (isFlap || state === "paused") flap();
  else pause();
});
canvas.addEventListener("pointerdown", event => {
  if (event.button !== 0) return;
  event.preventDefault();
  canvas.focus({ preventScroll:true });
  flap();
});
addEventListener("blur", pause);
document.addEventListener("visibilitychange", () => { if (document.hidden) pause(); });
reset();
if (ctx) requestAnimationFrame(loop);
else {
  canvas.hidden = true;
  statusEl.classList.remove("sr-only");
  statusEl.textContent = "This browser cannot draw the game. Try another browser.";
}
