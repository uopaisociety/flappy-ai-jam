// === AI PROMPT ZONE: CONFIG ===
// Positions are pixels, speed is pixels/second, gravity is pixels/second².
// Start with colours or one mechanic. The first gates teach the rhythm. Speed and gap difficulty then ramp over a minute.
const W = 400, H = 600;
const GRAVITY = 880, MAX_HOLE_GRAVITY = 1340, BLACK_HOLE_RAMP_SECONDS = 75;
const FLAP_STRENGTH = -285, MAX_FALL_SPEED = 400;
const BOOST_VELOCITY = -520, BOOST_GRAVITY = 380, BOOST_LIFT_SECONDS = 0.45;
const BOOST_SHIELD_SECONDS = 0.55, BOOST_COOLDOWN = 0.85, START_BOOSTS = 2;
const ORB_EFFECT_SECONDS = 10, GHOST_SPEED_MULT = 1.45, HEAVY_GRAVITY_MULT = 1.6, HEAVY_FALL_MULT = 1.35;
const LOSE_WEIGHT_MULT = 0.78, MIN_GRAVITY_FACTOR = 0.35, DEBRIS_WEIGHT_ADD = 0.28, MAX_GRAVITY_FACTOR = 2.2;
const JETTISON_COOLDOWN = 1.6, DEBRIS_PER_JETTISON = 3, EJECT_BONUS = 10;
const EJECT_BTN = { x: W - 102, y: 12, w: 90, h: 28 };
const DOUBLE_GAP_CHANCE = 0.45, DOUBLE_GAP_DIVIDER = 58, ORB_RADIUS = 13;
const WORMHOLE_CHANCE = 0.32, WORMHOLE_RADIUS = 20, WORMHOLE_FORWARD_GATES = 3, WORMHOLE_BACK_GATES = 2;
const MOVE_PIPE_SCORE = 30, PIPE_MOVE_AMP = 42, PIPE_MOVE_SPEED = 2.1;
const PIPE_GAP = 205, MIN_PIPE_GAP = 145, PIPE_WIDTH = 58, PIPE_SPACING = 260;
const PIPE_SPEED = 140, MAX_PIPE_SPEED = 220, RAMP_DELAY = 8, RAMP_SECONDS = 60;
const GROUND_H = 50, START_LIVES = 3, RECOVERY_SECONDS = 1.2, STEP = 1 / 120;
const COLORS = {
  navy:"#F8FAFC", cream:"#090710", card:"#120E1D", suit:"#F8FAFC", visor:"#F59E0B",
  boost:"#22C55E", orb:"#A855F7", ghost:"#38BDF8", heavy:"#F43F5E", light:"#10B981",
  malfunction:"#E11D48", eject:"#DC2626", wormhole:"#06B6D4", pipe:"#452710", cap:"#F59E0B",
  beam:"rgba(251, 191, 36, 0.22)", ground:"#171326"

const BOOK_PALETTE = ["#B45309", "#92400E", "#D97706", "#78350F", "#F59E0B", "#9A3412"];

let gravityFactor, gravityElapsed, jettisonCooldown, ejectedRun;

try {
  const stored = Number(localStorage.getItem("flappyAiJamBest"));
  if (Number.isSafeInteger(stored) && stored > 0) best = stored;

function resize() {
  const scale = Math.min((playArea.clientWidth - 24) / W, (playArea.clientHeight - 24) / H);
  if (scale <= 0) return;
  canvas.style.width = Math.floor(W * scale) + "px";
  canvas.style.height = Math.floor(H * scale) + "px";

function reset() {
  bird = { x:100, y:285, vy:0, r:15 };
  pipes = [];
  wormholes = [];
  debris = [];
  score = elapsed = gravityElapsed = recovery = boostCooldown = boostLift = boostShield = 0;
  ghostTimer = heavyTimer = malfunctionTimer = jettisonCooldown = noticeTimer = 0;
  postGhostShield = false;
  ejectedRun = false;
  blockedInput = null;
  lastInputMode = "keyboard";
  gravityFactor = 1;
  noticeText = "";
  noticeColor = COLORS.wormhole;
  lives = START_LIVES;
  boosts = START_BOOSTS;
  nextCenter = bird.y;
}

function showNotice(text, color, seconds = 2.4) {
  noticeText = text;
  noticeColor = color;
  noticeTimer = seconds;
}

function getHoleProximity() {
  return Math.min(1, Math.max(0, gravityElapsed / BLACK_HOLE_RAMP_SECONDS));
}

function getBaseGravity() {
  const pull = getHoleProximity();
  const holeGravity = GRAVITY + (MAX_HOLE_GRAVITY - GRAVITY) * pull;
  return holeGravity * gravityFactor;
}

function jettisonWeight() {
  gravityFactor = 1;
  gravityElapsed = 0;
  heavyTimer = 0;
  const playableH = H - GROUND_H;
  for (let i = 0; i < DEBRIS_PER_JETTISON; i++) {
    const spreadX = W + 45 + i * 95 + Math.random() * 35;
    const debY = Math.max(46, Math.min(playableH - 46, bird.y + (i - 1) * 78 + (Math.random() * 44 - 22)));
    debris.push({
      x: spreadX,
      y: debY,
      r: 12,
      vy: (Math.random() < 0.5 ? -1 : 1) * (24 + Math.random() * 26),
      rot: Math.random() * Math.PI * 2,
      spin: (Math.random() - 0.5) * 3.5,
      hit: false
    });
  }
  showNotice("JETTISONED · GRAVITY RESET", COLORS.light, 2.2);
  statusEl.textContent = "Weight jettisoned! Gravity reset, space debris ahead.";
}

function ejectPod() {
  if (state !== "playing") return;
  score += EJECT_BONUS;
  if (score > best) {
    best = score;
    try { localStorage.setItem("flappyAiJamBest", String(best)); } catch { /* Optional. */ }
  }
  ejectedRun = true;
  lives = 0;
  state = "over";
  overAt = performance.now();
  statusEl.textContent = "Ejected! +" + EJECT_BONUS + " bonus awarded. Final score " + score + ". Tap or press Space to play again.";
}

// === AI PROMPT ZONE: PHYSICS & INPUT ===
// Each press is one flap. Ignore a held key's repeat events.
// Controller malfunction forces switching between keyboard and tap/click.
// Pressing Space while playing also jettisons weight and spawns debris.
function flap(inputMode = "keyboard", isSpacebar = false) {
  if (!ctx || document.hidden) return;
  const freshStart = state !== "playing";
  if (state === "paused") {
    state = "playing";
  } else if (state === "over") {
    if (performance.now() - overAt < 350) return;
    reset();
    state = "playing";
    bird.vy = FLAP_STRENGTH;
  } else {
    if (state === "playing" && malfunctionTimer > 0 && blockedInput === inputMode) {
      statusEl.textContent = blockedInput === "keyboard"
        ? "Controller malfunction! Tap the screen to flap."
        : "Controller malfunction! Press Space or Up arrow to flap.";
      return;
    }
    state = "playing";
    bird.vy = FLAP_STRENGTH;
    if (!freshStart && isSpacebar && jettisonCooldown <= 0) {
      jettisonCooldown = JETTISON_COOLDOWN;
      jettisonWeight();
    }
  }
  lastInputMode = inputMode;
  if (freshStart) {
    lastTime = performance.now();
    accumulator = 0;
    statusEl.textContent = "Playing. " + lives + " lives, " + boosts + " boosts. Keep thrusting.";
  }
}
function boostJump() {
  if (!ctx || document.hidden || state !== "playing") return;
  if (boosts <= 0 || boostCooldown > 0) return;
  boosts--;
  boostCooldown = BOOST_COOLDOWN;
  boostLift = BOOST_LIFT_SECONDS;
  boostShield = Math.max(boostShield, BOOST_SHIELD_SECONDS);
  bird.vy = BOOST_VELOCITY;
  statusEl.textContent = "Boost jump! " + boosts + (boosts === 1 ? " boost left." : " boosts left.");
}
function pause() {
  if (state !== "playing") return;
  state = "paused";
  statusEl.textContent = "Paused. Tap or press Space to resume.";
  lastTime = null;
  accumulator = 0;
}
function updateBird(dt) {
  const baseGravity = getBaseGravity();
  let activeGravity = boostLift > 0 ? Math.max(BOOST_GRAVITY, baseGravity * 0.45) : baseGravity;
  let maxFall = MAX_FALL_SPEED * Math.max(0.75, Math.min(1.5, gravityFactor)) + getHoleProximity() * 85;
  if (heavyTimer > 0) {
    activeGravity *= HEAVY_GRAVITY_MULT;
    maxFall *= HEAVY_FALL_MULT;
  }
  bird.vy = Math.min(maxFall, bird.vy + activeGravity * dt);
  bird.y += bird.vy * dt;
  // The ceiling is forgiving. The floor uses the same life/recovery rule as pipes.
  if (bird.y < bird.r) {
    bird.y = bird.r;
    bird.vy = Math.max(0, bird.vy);
  }
  if (bird.y + bird.r > H - GROUND_H) {
    bird.y = H - GROUND_H - bird.r;
    bird.vy = 0;
    loseLife();
  }
}

// === AI PROMPT ZONE: PIPES / LEVEL ===
// The first two gaps are centered. Later gaps move farther as the run progresses.
// Some pipes split into two gaps with a mystery boost orb inside one of them.
// Wormholes occasionally spawn between pipes at the top or bottom, warping forward or back.
// When Ghost Speed ends, a safety shield stays active until aligned with a pillar gap.
// Pipes spawned at score 30+ move vertically within reachable bounds.
function createPipeAt(spawnX, difficulty) {
  const gap = PIPE_GAP + (MIN_PIPE_GAP - PIPE_GAP) * difficulty;
  const drift = 55 + 30 * difficulty;
  if (score > 0) nextCenter = Math.max(gap / 2 + 50,
    Math.min(H - GROUND_H - gap / 2 - 50, nextCenter + (Math.random() * 2 - 1) * drift));
  const playableH = H - GROUND_H;
  const subGap = Math.max(118, gap * 0.84);
  const minNeeded = subGap * 2 + DOUBLE_GAP_DIVIDER + 76;
  const canMove = score >= MOVE_PIPE_SCORE;
  let gaps, orb = null, minShift = 0, maxShift = 0;
  if (score >= 2 && playableH >= minNeeded && Math.random() < DOUBLE_GAP_CHANCE) {
    const top1 = 38 + Math.random() * Math.max(10, playableH - minNeeded);
    const top2 = top1 + subGap + DOUBLE_GAP_DIVIDER;
    const orbInTop = Math.random() < 0.5;
    const orbY = orbInTop ? top1 + subGap / 2 : top2 + subGap / 2;
    gaps = [
      { top: top1, size: subGap },
      { top: top2, size: subGap }
    ];
    orb = { y: orbY, r: ORB_RADIUS, collected: false };
    minShift = -(top1 - 24);
    maxShift = playableH - (top2 + subGap) - 24;
  } else {
    const top = nextCenter - gap / 2;
    gaps = [{ top, size: gap }];
    minShift = -(top - 24);
    maxShift = playableH - (top + gap) - 24;
  }
  const allowedAmp = canMove ? Math.min(PIPE_MOVE_AMP, Math.max(0, Math.min(-minShift, maxShift))) : 0;
  return {
    x: spawnX,
    gaps,
    orb,
    seed: Math.floor(Math.random() * 100),
    baseOffset: 0,
    moveAmp: allowedAmp,
    moveSpeed: PIPE_MOVE_SPEED,
    movePhase: Math.random() * Math.PI * 2,
    passed: false
  };
}
function triggerOrbBoost() {
  const roll = Math.random();
  if (roll < 0.25) {
    ghostTimer = ORB_EFFECT_SECONDS;
    postGhostShield = false;
    heavyTimer = 0;
    showNotice("GHOST SPEED · 10s", COLORS.ghost, 2.2);
    statusEl.textContent = "Ghost speed boost! Phase through matrix pillars for 10 seconds.";
  } else if (roll < 0.5) {
    gravityFactor = Math.max(MIN_GRAVITY_FACTOR, +(gravityFactor * LOSE_WEIGHT_MULT).toFixed(3));
    showNotice("LOSE WEIGHT · GRAVITY " + Math.round(gravityFactor * 100) + "%", COLORS.light, 2.6);
    statusEl.textContent = "Suit lightened! Gravity pull permanently reduced to " + Math.round(gravityFactor * 100) + "%.";
  } else if (roll < 0.75) {
    heavyTimer = ORB_EFFECT_SECONDS;
    ghostTimer = 0;
    showNotice("HEAVY GRAVITY · 10s", COLORS.heavy, 2.2);
    statusEl.textContent = "Heavy gravity! Gravity increased for 10 seconds.";
  } else {
    malfunctionTimer = ORB_EFFECT_SECONDS;
    blockedInput = lastInputMode;
    const switchMsg = blockedInput === "keyboard" ? "SWITCH TO TAP!" : "SWITCH TO SPACEBAR!";
    showNotice("MALFUNCTION · " + switchMsg, COLORS.malfunction, 2.8);
    statusEl.textContent = "Controller malfunction for 10 seconds! " + switchMsg;
  }
}
function triggerWormholeWarp(difficulty) {
  const warpForward = Math.random() < 0.5;
  const gates = warpForward ? WORMHOLE_FORWARD_GATES : WORMHOLE_BACK_GATES;
  const deltaX = (warpForward ? -1 : 1) * gates * PIPE_SPACING;
  for (const p of pipes) p.x += deltaX;
  for (const w of wormholes) w.x += deltaX;
  for (const d of debris) d.x += deltaX;

  if (warpForward) {
    for (const p of pipes) {
      if (!p.passed && p.x + PIPE_WIDTH < bird.x - bird.r) {
        p.passed = true;
        score++;
      }
    }
    if (score > best) {
      best = score;
      try { localStorage.setItem("flappyAiJamBest", String(best)); } catch { /* Optional. */ }
    }
    while (!pipes.length || pipes[pipes.length - 1].x <= W - PIPE_SPACING) {
      const nextX = pipes.length ? pipes[pipes.length - 1].x + PIPE_SPACING : W + 80;
      pipes.push(createPipeAt(nextX, difficulty));
    }
    showNotice("WARP FORWARD  +" + gates + " GATES", "#0284C7", 2.4);
    statusEl.textContent = "Wormhole warp! Shot ahead " + gates + " pillars.";
  } else {
    pipes = pipes.filter(p => p.x <= W + PIPE_SPACING * 2);
    let firstX = pipes.length ? pipes[0].x : W + 120;
    while (firstX - PIPE_SPACING > bird.x + 120) {
      firstX -= PIPE_SPACING;
      pipes.unshift(createPipeAt(firstX, difficulty));
    }
    showNotice("WARP BACKWARD  -" + gates + " GATES", "#D97706", 2.4);
    statusEl.textContent = "Wormhole rewind! Sent back " + gates + " pillars.";
  }

  boostShield = Math.max(boostShield, 1.15);
  const upcoming = pipes.find(item => item.x + PIPE_WIDTH >= bird.x - bird.r);
  if (upcoming && upcoming.gaps.length) {
    const targetGap = upcoming.gaps[0];
    bird.y = targetGap.top + targetGap.size / 2;
    bird.vy = Math.min(0, bird.vy);
  }
}
function applyPipeVerticalOffset(pipe, deltaY) {
  for (const g of pipe.gaps) g.top += deltaY;
  if (pipe.orb) pipe.orb.y += deltaY;
}
function isSafelyAlignedWithPillar() {
  const target = pipes.find(p => p.x + PIPE_WIDTH >= bird.x - bird.r);
  if (!target) return true;
  return target.gaps.some(g => bird.y - bird.r >= g.top + 4 && bird.y + bird.r <= g.top + g.size - 4);
}
function updatePipes(dt) {
  const difficulty = Math.min(1, Math.max(0, (elapsed - RAMP_DELAY) / RAMP_SECONDS));
  const baseSpeed = PIPE_SPEED + (MAX_PIPE_SPEED - PIPE_SPEED) * difficulty;
  const speed = baseSpeed * (ghostTimer > 0 ? GHOST_SPEED_MULT : 1);
  const last = pipes[pipes.length - 1];
  if (!last || last.x <= W - PIPE_SPACING) {
    const spawnX = last ? W : W + 100;
    const newPipe = createPipeAt(spawnX, difficulty);
    pipes.push(newPipe);
    if (last && score >= 2 && Math.random() < WORMHOLE_CHANCE) {
      const midX = (last.x + PIPE_WIDTH + spawnX) / 2;
      const atTop = Math.random() < 0.5;
      const playableH = H - GROUND_H;
      const wy = atTop ? 48 : playableH - 48;
      wormholes.push({
        x: midX,
        y: wy,
        r: WORMHOLE_RADIUS,
        atTop,
        used: false,
        spin: Math.random() * Math.PI * 2
      });
    }
  }

  // Drop the post-ghost shield once the astronaut is safely aligned with a pillar gap
  if (postGhostShield && isSafelyAlignedWithPillar()) {
    postGhostShield = false;
    statusEl.textContent = "Aligned with matrix gap. Re-entry shield disengaged.";
  }

  const phasing = recovery > 0 || boostShield > 0 || ghostTimer > 0 || postGhostShield;
  for (const w of wormholes) {
    w.x -= speed * dt;
    w.spin += dt * 4.5;
    if (!w.used) {
      const dx = bird.x - w.x;
      const dy = bird.y - w.y;
      const hitDist = bird.r + w.r - 2;
      if (dx * dx + dy * dy <= hitDist * hitDist) {
        w.used = true;
        triggerWormholeWarp(difficulty);
        break;
      }
    }
  }
  wormholes = wormholes.filter(w => !w.used && w.x + w.r + 12 > 0);

  const playableH = H - GROUND_H;
  for (const d of debris) {
    d.x -= speed * 1.12 * dt;
    d.y += d.vy * dt;
    d.rot += d.spin * dt;
    if (d.y < 28 || d.y > playableH - 28) d.vy *= -1;
    if (!d.hit) {
      const dx = bird.x - d.x;
      const dy = bird.y - d.y;
      const hitDist = bird.r + d.r - 2;
      if (dx * dx + dy * dy <= hitDist * hitDist) {
        d.hit = true;
        gravityFactor = Math.min(MAX_GRAVITY_FACTOR, +(gravityFactor + DEBRIS_WEIGHT_ADD).toFixed(3));
        showNotice("DEBRIS HIT · WEIGHT +" + Math.round(DEBRIS_WEIGHT_ADD * 100) + "%", "#EA580C", 2.0);
        statusEl.textContent = "Hit jettisoned debris! Gravity pull increased to " + Math.round(gravityFactor * 100) + "%.";
      }
    }
  }
  debris = debris.filter(d => !d.hit && d.x + d.r + 10 > 0);

  for (const pipe of pipes) {
    pipe.x -= speed * dt;
    if (pipe.moveAmp > 0) {
      const nextOffset = Math.sin(elapsed * pipe.moveSpeed + pipe.movePhase) * pipe.moveAmp;
      applyPipeVerticalOffset(pipe, nextOffset - pipe.baseOffset);
      pipe.baseOffset = nextOffset;
    }
    if (pipe.orb && !pipe.orb.collected) {
      const orbX = pipe.x + PIPE_WIDTH / 2;
      const dx = bird.x - orbX;
      const dy = bird.y - pipe.orb.y;
      const hitDist = bird.r + pipe.orb.r;
      if (dx * dx + dy * dy <= hitDist * hitDist) {
        pipe.orb.collected = true;
        triggerOrbBoost();
      }
    }
    const overlapsX = bird.x + bird.r - 3 > pipe.x && bird.x - bird.r + 3 < pipe.x + PIPE_WIDTH;
    const insideAnyGap = pipe.gaps.some(g => bird.y - bird.r + 3 >= g.top && bird.y + bird.r - 3 <= g.top + g.size);
    if (!phasing && overlapsX && !insideAnyGap) {
      loseLife(pipe);
      if (state === "over") return;
    }
    if (!pipe.passed && pipe.x + PIPE_WIDTH < bird.x - bird.r) {
      pipe.passed = true;
      score++;
      if (score > best) {
        best = score;
        try { localStorage.setItem("flappyAiJamBest", String(best)); } catch { /* Optional. */ }
      }
    }
  }
  pipes = pipes.filter(pipe => pipe.x + PIPE_WIDTH > 0);
}

// === AI PROMPT ZONE: SCORE & GAME OVER ===
// One contact costs one life. Recovery prevents overlapping obstacles taking all lives.
// Return immediately when the last life is lost. Never score after game over.
function loseLife(pipe) {
  if (recovery || state !== "playing") return;
  lives--;
  if (!lives) {
    state = "over";
    overAt = performance.now();
    statusEl.textContent = "Flight finished. Score " + score + ". Tap or press Space to try again.";
    return;
  }
  recovery = RECOVERY_SECONDS;
  const next = pipe || pipes.find(item => item.x + PIPE_WIDTH >= bird.x - bird.r);
  if (next && next.gaps.length) {
    let bestGap = next.gaps[0];
    for (const g of next.gaps) {
      if (Math.abs(bird.y - (g.top + g.size / 2)) < Math.abs(bird.y - (bestGap.top + bestGap.size / 2))) {
        bestGap = g;
      }
    }
    bird.y = bestGap.top + bestGap.size / 2;
  } else {
    bird.y = 285;
  }
  bird.vy = 0;
  statusEl.textContent = lives + (lives === 1 ? " life left." : " lives left.") + " Keep thrusting.";
}

function drawTesseractColumn(x, y, w, h, seed, capTop, capBottom) {
  if (h <= 0) return;
  ctx.fillStyle = "#231409";
  ctx.fillRect(x, y, w, h);

  const shelfStep = 26;
  const firstShelf = Math.ceil(y / shelfStep) * shelfStep;
  for (let sy = firstShelf; sy < y + h - 6; sy += shelfStep) {
    const rowIdx = Math.floor(sy / shelfStep) + seed;
    const bookH = Math.min(16, y + h - sy - 4);
    if (bookH > 4) {
      for (let bx = x + 6; bx < x + w - 10; bx += 9) {
        const colIdx = (rowIdx + Math.floor(bx)) % BOOK_PALETTE.length;
        ctx.fillStyle = BOOK_PALETTE[colIdx];
        ctx.fillRect(bx, sy + 3, 6, bookH);
      }
    }
    ctx.fillStyle = "rgba(251, 191, 36, 0.48)";
    ctx.fillRect(x + 3, sy, w - 6, 2);
  }

  ctx.strokeStyle = "#F59E0B";
  ctx.lineWidth = 2;
  ctx.strokeRect(x + 1, y + 1, w - 2, h - 2);
  ctx.strokeStyle = "rgba(254, 240, 138, 0.55)";
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.moveTo(x + 5, y);
  ctx.lineTo(x + 5, y + h);
  ctx.moveTo(x + w - 5, y);
  ctx.lineTo(x + w - 5, y + h);
  ctx.stroke();

  ctx.fillStyle = COLORS.cap;
  const capH = Math.min(12, h);
  if (capTop) ctx.fillRect(x - 2, y, w + 4, capH);
  if (capBottom) ctx.fillRect(x - 2, y + h - capH, w + 4, capH);
}

  } else if (state === "paused") {

// === AI PROMPT ZONE: GAME LOOP ===
// Fixed steps make the same inputs play the same at 30, 60 and 144Hz.
// Clamp long frames and pause on focus loss so a tab switch cannot cost a life.
function loop(timestamp) {
  if (lastTime !== null && state === "playing") {
    accumulator += Math.min(0.05, Math.max(0, (timestamp - lastTime) / 1000));
    while (accumulator + 1e-9 >= STEP && state === "playing") {
      recovery = Math.max(0, recovery - STEP);
      boostCooldown = Math.max(0, boostCooldown - STEP);
      boostLift = Math.max(0, boostLift - STEP);
      boostShield = Math.max(0, boostShield - STEP);
      jettisonCooldown = Math.max(0, jettisonCooldown - STEP);
      if (ghostTimer > 0) {
        ghostTimer = Math.max(0, ghostTimer - STEP);
        if (ghostTimer === 0 && !isSafelyAlignedWithPillar()) {
          postGhostShield = true;
          statusEl.textContent = "Ghost speed ended. Alignment shield active until safe.";
        }
      }
      heavyTimer = Math.max(0, heavyTimer - STEP);
      if (malfunctionTimer > 0) {
        malfunctionTimer = Math.max(0, malfunctionTimer - STEP);
        if (malfunctionTimer === 0) {
          blockedInput = null;
          statusEl.textContent = "Controls recalibrated.";
        }
      }
      noticeTimer = Math.max(0, noticeTimer - STEP);
      elapsed += STEP;
      gravityElapsed += STEP;
      updateBird(STEP);
      if (state === "playing") updatePipes(STEP);
      accumulator = Math.max(0, accumulator - STEP);
    }
  }
  lastTime = timestamp;
  drawBackground(); drawWormholes(); drawDebris(); drawPipes(); drawBird(); drawUI();
  requestAnimationFrame(loop);
}

  if (event.defaultPrevented || event.altKey || event.ctrlKey || event.metaKey) return;
  if (event.target.closest?.("a, button, input, textarea, select, [contenteditable]")) return;
  const isSpace = event.code === "Space";
  const isFlap = isSpace || event.code === "ArrowUp";
  const isBoost = event.code === "ShiftLeft" || event.code === "ShiftRight" || event.code === "ArrowDown";
  const isEject = event.code === "KeyE";
  if (!isFlap && !isBoost && !isEject && event.code !== "KeyP") return;
  event.preventDefault();
  if (event.repeat) return;
  if (isEject) ejectPod();
  else if (isBoost) boostJump();
  else if (isFlap || state === "paused") flap("keyboard", isSpace);
  else pause();
});
