// === AI PROMPT ZONE: CONFIG ===
// Positions are pixels, speed is pixels/second, gravity is pixels/second².
// Start with colours or one mechanic. The first gates teach the rhythm. Speed and gap difficulty then ramp over a minute.
// STAGES_PER_BOSS is how many creature gates you pass before each boss. Set it to 3 to test the boss fights quickly.
// Each boss fight is BOSS_GATES moving gates. Every boss defeated makes later boss fights BOSS_SPEEDUP (2%) faster.
// Asclepius restores one life after Hades and Poseidon, capped at three lives.
const W = 400, H = 600;
const GRAVITY = 880, FLAP_STRENGTH = -285, MAX_FALL_SPEED = 400;
const PIPE_GAP = 205, MIN_PIPE_GAP = 145, PIPE_WIDTH = 58, PIPE_SPACING = 260;
const PIPE_SPEED = 140, MAX_PIPE_SPEED = 220, RAMP_DELAY = 8, RAMP_SECONDS = 60;
const GROUND_H = 50, START_LIVES = 3, MAX_LIVES = 3, RECOVERY_SECONDS = 1.2, STEP = 1 / 120;
const STAGES_PER_BOSS = 40, BOSS_GATES = 12, BOSS_SPEEDUP = 0.02;
const BOSS_SPEED = 190, BOSS_GAP = 170, BOSS_AMP = 50, BOSS_PERIOD = 2.6, HEAL_SECONDS = 3.5;
const HEAD_H = 34;
const COLORS = { navy:"#073B5C", cream:"#F7F5F0", bird:"#FFC93C", beak:"#FF7A00", pipe:"#0E5A8A", ground:"#C9B79C",

// Body colour, darker scale colour and face colour for each creature: Hydra, Cyclops, Medusa, Minotaur, Cerberus.

  { body:"#2E8B57", dark:"#1F6B40", face:"#3FA46A" },
  { body:"#B07A52", dark:"#8A5A38", face:"#C48F66" },

];

  const stored = Number(localStorage.getItem("flappyAiJamBest"));
  if (Number.isSafeInteger(stored) && stored > 0) best = stored;

  const scale = Math.min((playArea.clientWidth - 24) / W, (playArea.clientHeight - 24) / H);
  if (scale <= 0) return;
  canvas.style.width = Math.floor(W * scale) + "px";
  canvas.style.height = Math.floor(H * scale) + "px";

function reset() {
  bird = { x:100, y:285, vy:0, r:15 };
  pipes = [];
  score = elapsed = recovery = 0;
  spawned = 0;
  phase = "normal";
  phaseGates = 0;
  stageTime = 0;
  defeated = 0;
  bossIdx = 0;
  bossHp = BOSS_GATES;

// === AI PROMPT ZONE: PHYSICS & INPUT ===
// Each press is one flap. Ignore a held key's repeat events.
// A resume input leaves the current position intact. A restart creates a new run.
function flap() {
  if (!ctx || document.hidden) return;
  const freshStart = state !== "playing";
  if (state === "paused") {
    state = "playing";
  } else {
    if (state === "over" || state === "won") {
      if (performance.now() - overAt < 350) return;
      reset();
    }
    state = "playing";
    bird.vy = FLAP_STRENGTH;
  }
  if (freshStart) {
    lastTime = performance.now();
    accumulator = 0;
    statusEl.textContent = "Playing. " + lives + " lives. Keep flapping.";
  }
}
function pause() {
  if (state !== "playing") return;
  state = "paused";
  statusEl.textContent = "Paused. Tap or press Space to resume.";
  lastTime = null;
  accumulator = 0;
}
function updateBird(dt) {
  bird.vy = Math.min(MAX_FALL_SPEED, bird.vy + GRAVITY * dt);
  bird.y += bird.vy * dt;
  // The ceiling is forgiving. The floor uses the same life/recovery rule as the creatures.
  if (bird.y < bird.r) {
    bird.y = bird.r;
    bird.vy = Math.max(0, bird.vy);
  }
  if (bird.y + bird.r > H - GROUND_H) {
    bird.y = H - GROUND_H - bird.r;
    bird.vy = 0;
    loseLife();
  }
}

// === AI PROMPT ZONE: PIPES / LEVEL ===
// Each "pipe" is a pair of creatures with a gap between them. The first two gaps are centered.
// Later gaps move farther as the stage progresses. Existing gaps stay still while the player approaches,
// except in boss fights, where every new gap slides up and down. Boss fights get 2% faster per boss defeated.
// Every visible creature head belongs to its collision box.
function currentGod() { return GODS[Math.min(bossIdx, GODS.length - 1)]; }
function startBoss() {
  phase = "boss";
  bossIdx = defeated;
  bossHp = BOSS_GATES;
  bossTime = 0;
  healTime = 0;
  statusEl.textContent = currentGod().name + " appears! The creatures are moving. Slip between them.";
}
function defeatBoss() {
  const name = currentGod().name;
  defeated++;
  if (defeated >= GODS.length) { win(); return; }
  phase = "normal";
  phaseGates = 0;
  stageTime = 0;
  healGained = lives < MAX_LIVES;
  lives = Math.min(MAX_LIVES, lives + 1);
  healTime = HEAL_SECONDS;
  statusEl.textContent = name + " is defeated! Asclepius " + (healGained ? "restores a life. " + lives + " lives." : "heals you. Lives are full.");
}
function updatePipes(dt) {
  const boss = phase === "boss";
  const mult = 1 + BOSS_SPEEDUP * defeated;
  const difficulty = Math.min(1, Math.max(0, (stageTime - RAMP_DELAY) / RAMP_SECONDS));
  const speed = boss ? BOSS_SPEED * mult : PIPE_SPEED + (MAX_PIPE_SPEED - PIPE_SPEED) * difficulty;
  const last = pipes[pipes.length - 1];
  if (!last || last.x <= W - PIPE_SPACING) {
    const gap = boss ? BOSS_GAP : PIPE_GAP + (MIN_PIPE_GAP - PIPE_GAP) * difficulty;
    const drift = boss ? 70 : 55 + 30 * difficulty;
    const margin = boss ? 40 + BOSS_AMP : 50;
    if (score > 0) nextCenter = Math.max(gap / 2 + margin,
      Math.min(H - GROUND_H - gap / 2 - margin, nextCenter + (Math.random() * 2 - 1) * drift));
    pipes.push({ x:last ? W : W + 100, top:nextCenter - gap / 2, gap, passed:false,
      base:nextCenter, phase:spawned * 1.3, period:BOSS_PERIOD / mult, moving:boss,
      kind:spawned % 5, kind2:(spawned + 2) % 5 });
    spawned++;
  }
  for (const pipe of pipes) {
    pipe.x -= speed * dt;
    if (pipe.moving) {
      pipe.top = pipe.base + BOSS_AMP * Math.sin(elapsed * Math.PI * 2 / pipe.period + pipe.phase) - pipe.gap / 2;
    }
    const overlaps = bird.x + bird.r - 3 > pipe.x && bird.x - bird.r + 3 < pipe.x + PIPE_WIDTH;
    const outsideGap = bird.y - bird.r + 3 < pipe.top || bird.y + bird.r > pipe.top + pipe.gap + 3;
    if (!recovery && overlaps && outsideGap) {
      loseLife(pipe);
      if (state === "over") return;
    }
    if (!pipe.passed && pipe.x + PIPE_WIDTH < bird.x - bird.r) {
      pipe.passed = true;
      score++;
      // Athena has a 1-in-1,000 chance on each passed pipe to grant a life.
      // She only appears when a life can be added; the three-life cap still applies.
      if (lives < MAX_LIVES && Math.random() < 1 / 1000) {
        lives = Math.min(MAX_LIVES, lives + 1);
        athenaTime = 3.5;
        athenaGained = true;
        statusEl.textContent = "Athena has blessed you! +1 life. " + lives + " lives.";
      }
      if (score > best) {
        best = score;
        try { localStorage.setItem("flappyAiJamBest", String(best)); } catch { /* Optional. */ }
      }
      if (boss && pipe.moving) {
        bossHp--;
        if (bossHp <= 0) {
          defeatBoss();
          if (state === "won") return;
        }
      } else if (!boss) {
        phaseGates++;
        if (phaseGates >= STAGES_PER_BOSS) startBoss();
      }
    }
  }
  pipes = pipes.filter(pipe => pipe.x + PIPE_WIDTH > 0);
}

// === AI PROMPT ZONE: SCORE & GAME OVER ===
// One contact costs one life. Recovery prevents overlapping obstacles taking all lives.
// Return immediately when the last life is lost. Never score after game over.
// Passing BOSS_GATES moving gates defeats a boss. Defeating Zeus wins the run.
function loseLife(pipe) {
  if (recovery || state !== "playing") return;
  lives--;
  if (!lives) {
    state = "over";
    overAt = performance.now();
    statusEl.textContent = "Flight finished. Score " + score + ". Tap or press Space to try again.";
    return;
  }
  recovery = RECOVERY_SECONDS;
  const next = pipe || pipes.find(item => item.x + PIPE_WIDTH >= bird.x - bird.r);
  bird.y = next ? next.top + next.gap / 2 : 285;
  bird.vy = 0;
  statusEl.textContent = lives + (lives === 1 ? " life left." : " lives left.") + " Keep flapping.";
}
function win() {
  state = "won";
  overAt = performance.now();
  statusEl.textContent = "Olympus conquered! Zeus is defeated. Score " + score + ". Tap or press Space to play again.";
}

// === AI PROMPT ZONE: GRAPHICS ===
// Draw with shapes. Keep controls and the score legible over any new artwork.
function rr(x, y, w, h, r) {
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.arcTo(x + w, y, x + w, y + h, r);
  ctx.arcTo(x + w, y + h, x, y + h, r);
  ctx.arcTo(x, y + h, x, y, r);
  ctx.arcTo(x, y, x + w, y, r);
  ctx.closePath();
}
function dot(x, y, r, color) {
  ctx.fillStyle = color;
  ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI * 2); ctx.fill();
}
// Draws the held item with its origin at the god's right hand.
function drawItem(item, c) {
  const t = elapsed;
  ctx.save();
  ctx.translate(64, 178);
  ctx.strokeStyle = c; ctx.fillStyle = c; ctx.lineWidth = 4; ctx.lineCap = "round";
  const staff = (top) => { ctx.beginPath(); ctx.moveTo(0, 142); ctx.lineTo(0, top); ctx.stroke(); };
  if (item === "bident" || item === "trident") {
    staff(-56);
    ctx.beginPath(); ctx.moveTo(-12, -76); ctx.lineTo(-12, -52); ctx.lineTo(12, -52); ctx.lineTo(12, -76); ctx.stroke();
    if (item === "trident") { ctx.beginPath(); ctx.moveTo(0, -84); ctx.lineTo(0, -52); ctx.stroke(); }
  } else if (item === "scepter") {
    staff(-40); dot(0, -50, 10, c); dot(-3, -53, 3, "#fff");
  } else if (item === "wheat") {
    staff(-30);

// === AI PROMPT ZONE: GAME LOOP ===
// Fixed steps make the same inputs play the same at 30, 60 and 144Hz.
// Clamp long frames and pause on focus loss so a tab switch cannot cost a life.
function loop(timestamp) {
  if (lastTime !== null && state === "playing") {
    accumulator += Math.min(0.05, Math.max(0, (timestamp - lastTime) / 1000));
    while (accumulator + 1e-9 >= STEP && state === "playing") {
      recovery = Math.max(0, recovery - STEP);
      elapsed += STEP;
      if (phase === "boss") bossTime += STEP; else stageTime += STEP;
      fade = Math.max(0, Math.min(1, fade + (phase === "boss" ? STEP : -STEP)));
      if (healTime > 0) healTime = Math.max(0, healTime - STEP);
      if (athenaTime > 0) athenaTime = Math.max(0, athenaTime - STEP);
      updateBird(STEP);
      if (state === "playing") updatePipes(STEP);
      accumulator = Math.max(0, accumulator - STEP);
    }
  }
  lastTime = timestamp;
  drawBackground(); drawPipes(); drawBird(); drawUI();
  requestAnimationFrame(loop);
}
addEventListener("keydown", event => {
  if (event.defaultPrevented || event.altKey || event.ctrlKey || event.metaKey) return;
  if (event.target.closest?.("a, button, input, textarea, select, [contenteditable]")) return;
  const isFlap = event.code === "Space" || event.code === "ArrowUp";
  if (!isFlap && event.code !== "KeyP") return;
  event.preventDefault();
  if (event.repeat) return;
  if (isFlap || state === "paused") flap();
  else pause();
});
