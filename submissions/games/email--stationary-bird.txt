"use strict";

// === AI PROMPT ZONE: CONFIG ===
// Positions are pixels, speed is pixels/second, gravity is pixels/second².
// Start with colours or one mechanic. The first gates teach the rhythm. Speed and gap difficulty then ramp over a minute.
const W = 400, H = 600, BIRD_SCREEN_Y = 285;
const GRAVITY = 880, FLAP_STRENGTH = -285, MAX_FALL_SPEED = 400;
const PIPE_GAP = 180, MIN_PIPE_GAP = 130, PIPE_WIDTH = 58, PIPE_SPACING = 260;
const PIPE_SPEED = 140, MAX_PIPE_SPEED = 220, RAMP_DELAY = 8, RAMP_SECONDS = 60;
const WIND_WARNING_SECONDS = 0.75, WIND_PIPE_SPEED_FACTOR = 0.45;
const START_LIVES = 3, RECOVERY_SECONDS = 1.2, BONUS_FEEDBACK_SECONDS = 1, STEP = 1 / 120;
const COLORS = { navy:"#073B5C", cream:"#F7F5F0", bird:"#FFC93C", beak:"#FF7A00", pipe:"#0E5A8A", bonusTip:"#C62828" };
const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");
const statusEl = document.getElementById("status");
let state = "start", bird, pipes, score, lives, recovery, elapsed, nextCenter, pipesSpawned, nextBonusPipe;
let windDirection, windWarning, windRemaining, nextWindIn;
let bonusFeedbackTimer = 0;
let overAt = 0, best = 0, lastTime = null, accumulator = 0;
try {
  const stored = Number(localStorage.getItem("flappyAiJamBest"));
  if (Number.isSafeInteger(stored) && stored > 0) best = stored;
} catch { /* The game also works when browser storage is blocked. */ }

function resize() {
  const space = document.querySelector("main");
  const scale = Math.max(0.01, Math.min((space.clientWidth - 24) / W, (space.clientHeight - 24) / H));
  canvas.style.width = Math.floor(W * scale) + "px";
  canvas.style.height = Math.floor(H * scale) + "px";
}
addEventListener("resize", resize);
resize();

function reset() {
  bird = { x:100, y:BIRD_SCREEN_Y, vy:0, r:15 };
  pipes = [];
  score = elapsed = recovery = 0;
  lives = START_LIVES;
  nextCenter = bird.y;
  pipesSpawned = 0;
  nextBonusPipe = 3 + Math.floor(Math.random() * 3);
  bonusFeedbackTimer = 0;
  windDirection = windWarning = windRemaining = 0;
  nextWindIn = 2.5 + Math.random() * 3.5;
}

// === AI PROMPT ZONE: PHYSICS & INPUT ===
// Each press is one flap. Ignore a held key's repeat events.
// A resume input leaves the current position intact. A restart creates a new run.
function flap() {
  if (!ctx || document.hidden) return;
  const freshStart = state !== "playing";
  if (state === "paused") {
    state = "playing";
  } else {
    if (state === "over") {
      if (performance.now() - overAt < 350) return;
      reset();
    }
    state = "playing";
    bird.vy = FLAP_STRENGTH;
  }
  if (freshStart) {
    lastTime = performance.now();
    accumulator = 0;
    statusEl.textContent = "Playing. " + lives + " lives. Keep flapping.";
  }
}
function pause() {
  if (state !== "playing") return;
  state = "paused";
  statusEl.textContent = "Paused. Tap or press Space to resume.";
  lastTime = null;
  accumulator = 0;
}
function updateWind(dt) {
  if (windRemaining > 0) {
    windRemaining = Math.max(0, windRemaining - dt);
    if (!windRemaining) {
      windDirection = 0;
      nextWindIn = 3 + Math.random() * 4;
    }
  } else if (windWarning > 0) {
    windWarning = Math.max(0, windWarning - dt);
    if (!windWarning) windRemaining = 1.2 + Math.random() * 1.2;
  } else {
    nextWindIn -= dt;
    if (nextWindIn <= 0) {
      windDirection = Math.random() < 0.5 ? -1 : 1;
      windWarning = WIND_WARNING_SECONDS;
    }
  }
}
function updateBird(dt) {
  bird.vy = Math.min(MAX_FALL_SPEED, bird.vy + GRAVITY * dt);
  bird.y += bird.vy * dt;
  // Keep the bird from flying above the world; there is no floor collision.
  if (bird.y < bird.r) {
    bird.y = bird.r;
    bird.vy = Math.max(0, bird.vy);
  }
}

// === AI PROMPT ZONE: PIPES / LEVEL ===
// The first two gaps are centered. Later gaps move farther as the run progresses.
// Store each new gap on its pipe. Existing gaps stay still while the player approaches.
// Every visible pipe endcap belongs to its collision box.
function updatePipes(dt) {
  const difficulty = Math.min(1, Math.max(0, (elapsed - RAMP_DELAY) / RAMP_SECONDS));
  const speed = (PIPE_SPEED + (MAX_PIPE_SPEED - PIPE_SPEED) * difficulty)
    * (windRemaining > 0 ? WIND_PIPE_SPEED_FACTOR : 1);
  const last = pipes[pipes.length - 1];
  if (!last || last.x <= W - PIPE_SPACING) {
    const gap = PIPE_GAP + (MIN_PIPE_GAP - PIPE_GAP) * difficulty;
    const drift = 55 + 30 * difficulty;
    if (score > 0) nextCenter = Math.max(gap / 2 + 50,
      Math.min(H - gap / 2 - 50, nextCenter + (Math.random() * 2 - 1) * drift));
    pipesSpawned++;
    const bonus = pipesSpawned === nextBonusPipe;
    if (bonus) nextBonusPipe += 3 + Math.floor(Math.random() * 3);
    pipes.push({ x:last ? W : W + 100, top:nextCenter - gap / 2, gap, passed:false, bonus, bonusCollected:false });
  }
  for (const pipe of pipes) {
    pipe.x -= speed * dt;
    const overlaps = bird.x + bird.r - 3 > pipe.x && bird.x - bird.r + 3 < pipe.x + PIPE_WIDTH;
    const outsideGap = bird.y - bird.r + 3 < pipe.top || bird.y + bird.r - 3 > pipe.top + pipe.gap;
    const hitsBonusTip = pipe.bonus && overlaps && touchesPipeTip(pipe);
    if (hitsBonusTip) {
      if (!pipe.bonusCollected) {
        pipe.bonusCollected = true;
        awardPoints(1);
        bonusFeedbackTimer = BONUS_FEEDBACK_SECONDS;
      }
    } else if (!recovery && overlaps && outsideGap) {
      loseLife(pipe);
      if (state === "over") return;
    }
    if (!pipe.passed && pipe.x + PIPE_WIDTH < bird.x - bird.r) {
      pipe.passed = true;
      awardPoints(1);
    }
  }
  pipes = pipes.filter(pipe => pipe.x + PIPE_WIDTH > 0);
}

// === AI PROMPT ZONE: SCORE & GAME OVER ===
// Ordinary pillar contact costs one life; bonus tips reward contact without a life loss.
// Return immediately when the last life is lost. Never score after game over.
function touchesPipeTip(pipe) {
  const birdTop = bird.y - bird.r + 3;
  const birdBottom = bird.y + bird.r - 3;
  const lowerTipTop = pipe.top + pipe.gap;
  return (birdBottom > pipe.top - 16 && birdTop < pipe.top)
    || (birdBottom > lowerTipTop && birdTop < lowerTipTop + 16);
}
function awardPoints(points) {
  score += points;
  if (score > best) {
    best = score;
    try { localStorage.setItem("flappyAiJamBest", String(best)); } catch { /* Optional. */ }
  }
}
function loseLife(pipe) {
  if (recovery || state !== "playing") return;
  lives--;
  if (!lives) {
    state = "over";
    overAt = performance.now();
    statusEl.textContent = "Flight finished. Score " + score + ". Tap or press Space to try again.";
    return;
  }
  recovery = RECOVERY_SECONDS;
  const next = pipe || pipes.find(item => item.x + PIPE_WIDTH >= bird.x - bird.r);
  bird.y = next ? next.top + next.gap / 2 : 285;
  bird.vy = 0;
  statusEl.textContent = lives + (lives === 1 ? " life left." : " lives left.") + " Keep flapping.";
}

// === AI PROMPT ZONE: GRAPHICS ===
// Draw with shapes. Keep controls and the score legible over any new artwork.
// The bird stays anchored on screen while the world follows its simulated flight.
function drawBackground() {
  const worldOffset = BIRD_SCREEN_Y - bird.y;
  ctx.fillStyle = COLORS.cream;
  ctx.fillRect(0, 0, W, H);
  ctx.fillStyle = "rgba(7,59,92,0.08)";
  for (let i = 0; i < 4; i++) {
    const x = ((i * 140 - elapsed * 24) % (W + 120) + W + 120) % (W + 120) - 60;
    ctx.beginPath();
    ctx.arc(x, 80 + i * 45 + worldOffset, 30, 0, Math.PI * 2);
    ctx.arc(x + 30, 75 + i * 45 + worldOffset, 22, 0, Math.PI * 2);
    ctx.fill();
  }
}
function drawWindWarning() {
  if (!windWarning) return;
  const progress = 1 - windWarning / WIND_WARNING_SECONDS;
  ctx.save();
  ctx.strokeStyle = "rgba(14,90,138," + (0.3 + progress * 0.5) + ")";
  ctx.lineWidth = 2;
  ctx.lineCap = "round";
  for (let i = 0; i < 3; i++) {
    const x = bird.x - windDirection * (38 + i * 15) + windDirection * progress * 16;
    const y = BIRD_SCREEN_Y - 42 + i * 16;
    ctx.beginPath();
    ctx.moveTo(x, y);
    ctx.quadraticCurveTo(x + windDirection * 8, y - 5, x + windDirection * 18, y - 3);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(x + windDirection * 3, y + 4);
    ctx.lineTo(x + windDirection * 19, y + 2);
    ctx.stroke();
  }
  ctx.restore();
}
function drawPipes() {
  const worldOffset = BIRD_SCREEN_Y - bird.y;
  for (const pipe of pipes) {
    const top = pipe.top + worldOffset;
    const bottom = top + pipe.gap;
    ctx.fillStyle = COLORS.pipe;
    ctx.fillRect(pipe.x, 0, PIPE_WIDTH, Math.max(0, top));
    ctx.fillRect(pipe.x, bottom, PIPE_WIDTH, Math.max(0, H - bottom));
    ctx.fillStyle = pipe.bonus ? COLORS.bonusTip : COLORS.navy;
    ctx.fillRect(pipe.x, top - 16, PIPE_WIDTH, 16);
    ctx.fillRect(pipe.x, bottom, PIPE_WIDTH, 16);
  }
}
function drawBird() {
  ctx.save();
  ctx.translate(bird.x, BIRD_SCREEN_Y);
  ctx.rotate(Math.max(-0.4, Math.min(0.8, bird.vy / 500)));
  if (recovery) {
    ctx.strokeStyle = COLORS.navy;
    ctx.lineWidth = 2;
    ctx.beginPath(); ctx.arc(0, 0, bird.r + 7, 0, Math.PI * 2); ctx.stroke();
  }
  ctx.fillStyle = COLORS.bird;
  ctx.beginPath(); ctx.arc(0, 0, bird.r, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = "#fff";
  ctx.beginPath(); ctx.arc(6, -5, 6, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = COLORS.navy;
  ctx.beginPath(); ctx.arc(8, -5, 3, 0, Math.PI * 2); ctx.fill();
  ctx.fillStyle = COLORS.beak;
  ctx.beginPath(); ctx.moveTo(12, 0); ctx.lineTo(24, 4); ctx.lineTo(12, 8); ctx.fill();
  ctx.restore();
}
function drawText(text, y, size, color = COLORS.navy) {
  ctx.fillStyle = color;
  ctx.font = "bold " + size + "px system-ui, sans-serif";
  ctx.textAlign = "center";
  ctx.fillText(String(text), W / 2, y);
}
function drawUI() {
  drawText(score, 58, 38);
  drawText("Lives " + lives + " / " + START_LIVES + "   ·   Best " + best, H - 18, 16);
  if (state === "playing") {
    if (bonusFeedbackTimer > 0) drawText("+1!", 102, 24, COLORS.bonusTip);
    return;
  }
  ctx.fillStyle = COLORS.cream;
  ctx.fillRect(24, 175, W - 48, 205);
  ctx.strokeStyle = COLORS.navy; ctx.lineWidth = 2;
  ctx.strokeRect(24, 175, W - 48, 205);
  if (state === "start") {
    drawText("FLAPPY AI JAM", 225, 32);
    drawText("Bird stays still; flap shifts course", 260, 15);
    drawText("Gusts slow pillars; streaks warn", 290, 14);
    drawText("Touch red tips for +1 point", 320, 15);
    drawText("Tip contact won't cost a life", 350, 14);
  } else if (state === "paused") {
    drawText("PAUSED", 235, 32);
    drawText("Tap or press Space to resume", 305, 18);
  } else {
    drawText("FLIGHT OVER", 225, 32);
    drawText("Score " + score + "   ·   Best " + best, 275, 22);
    drawText("Tap or press Space to try again", 330, 18);
  }
}

// === AI PROMPT ZONE: GAME LOOP ===
// Fixed steps make the same inputs play the same at 30, 60 and 144Hz.
// Clamp long frames and pause on focus loss so a tab switch cannot cost a life.
function loop(timestamp) {
  if (lastTime !== null && state === "playing") {
    accumulator += Math.min(0.05, Math.max(0, (timestamp - lastTime) / 1000));
    while (accumulator + 1e-9 >= STEP && state === "playing") {
      recovery = Math.max(0, recovery - STEP);
      bonusFeedbackTimer = Math.max(0, bonusFeedbackTimer - STEP);
      elapsed += STEP;
      updateWind(STEP);
      updateBird(STEP);
      if (state === "playing") updatePipes(STEP);
      accumulator = Math.max(0, accumulator - STEP);
    }
  }
  lastTime = timestamp;
  drawBackground(); drawWindWarning(); drawPipes(); drawBird(); drawUI();
  requestAnimationFrame(loop);
}
addEventListener("keydown", event => {
  if (event.defaultPrevented || event.altKey || event.ctrlKey || event.metaKey) return;
  if (event.target.closest?.("a, button, input, textarea, select, [contenteditable]")) return;
  const isFlap = event.code === "Space" || event.code === "ArrowUp";
  if (!isFlap && event.code !== "KeyP") return;
  event.preventDefault();
  if (event.repeat) return;
  if (isFlap || state === "paused") flap();
  else pause();
});

  if (event.button !== 0) return;
  event.preventDefault();
  canvas.focus({ preventScroll:true });
  flap();
});
addEventListener("blur", pause);
document.addEventListener("visibilitychange", () => { if (document.hidden) pause(); });
reset();
if (ctx) requestAnimationFrame(loop);
